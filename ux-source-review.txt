<script>
import vibrator from "@system.vibrator";
import device from '@system.device'
import { SimpleInputMethod } from "./assets/dicUtil.js";
function doSearchDic(word, lang, cb) {
  if (!word) {
    cb([]);
    return;
  }

  const result = SimpleInputMethod.getHanzi(word, lang);
  cb(Array.isArray(result) && result[0] ? result[0] : []);
}

function deleteLast(t) {
  if (t) {
    return t.substr(0, t.length - 1);
  }
  return "";
}
export default {
  props: {
    hide: {
      default: true,
    },
    keyboardtype: {
      default: "QWERTY",
    },
    maxlength: {
      default: 5,
    },
    englishseed: {
      default: "",
    },
    englishsuggestions: {
      default: "",
    },
    vibratemode: {
      default: "",
    },
    screentype: {
      default: "circle",
    },
  },
  data: {
    cval: "",
    resultList: [],
    resultList2: [],
    waitingList: [],
    waitingIndex: -1,
    lastWaitingStr: "",
    downFlag: "",
    lang: "cn",
    numFlag: false,
    numFlag_jp: false,
    upperFlag: false,
    cvalList: [0, 1, 2, 3, 4],
    percent67: 52,
    percent66: 0,
    // 针对screenShape为rect的设备，会自动获取screenWidth并绑定到根div
    // 这样便能同时适配n67和o65甚至是后续设备，但实际效果可能受designWidth影响
    screenWidth: 336,
    keys: {
      full: [
        ["Q", "W", "E", "R", "T", "Y", "U", "I", "O", "P"],
        ["A", "S", "D", "F", "G", "H", "J", "K", "L"],
        ["Z", "X", "C", "V", "B", "N", "M"],
      ],
      sign: [

        ["~", "!", "@", "#", "%", "“", "”", "*", "?", "/"],
        ["(", ")", "-", "_", ":", ";", "，", "。", "."],
      ],
      sign_jp: [
        ["1", "2", "3", "4", "5", "6", "7", "8", "9", "0"],
        ["~", "•", "@", "#", "%", "「", "」", "*", "?", "/"],
        ["(", ")", "-", "…", ":", ";", "、", "。", "!"],
      ],
      sign62: [
        ["2", "3", "4", "5", "6", "7", "8", "9"],
        ["!", "@", "#", "%", "“", "”", "*"],
        [")", "-", "_", ":", ";"],
      ],
      sign62_jp: [
        ["2", "3", "4", "5", "6", "7", "8", "9"],
        ["•", "@", "#", "%", "「", "」", "*"],
        [")", "-", "…", ":", ";"],

      full62: [
        ["W", "E", "R", "T", "Y", "U", "I", "O"],
        ["S", "D", "F", "G", "H", "J", "K"],
        ["X", "C", "V", "B", "N"],
      ],
      t9: [
        ["abc", "def"],
        ["ghi", "jkl", "mno"],

  onInit() {
    if (this.lang !== "cn" && this.lang !== "en") {
      this.lang = "cn";
      this.numFlag_jp = false;
    }
    if (this.maxlength) {
      const tempCvalList = [];
      for (let i = 0; i < this.maxlength; i++) {
        tempCvalList.push(i);

      this.adjustScreenWidth();

    this.onVibrate();
    let content = txt;
    if (this.lang === "en") {
      const seed = this.normalizeEnglishSeed(this.englishseed);
      const word = this.normalizeEnglishSeed(txt);
      if (seed && word.indexOf(seed) === 0 && word.length > seed.length) {
        content = word.slice(seed.length);
      }
    }
    this.cval = "";
    this.addAllTxt(content);
    this.clearWaiting();
    this.resetReslutList();
    this.downFlag = "";
  },
  onBtnClick(sign) {
    this.onVibrate();
    switch (sign) {
      case "AC":
        this.cval = "";
        this.clearWaiting();
        this.resetReslutList();
        break;
      case "lang":
        this.lang = this.lang === "cn" ? "en" : "cn";
        this.cval = "";
        this.clearWaiting();
        this.resetReslutList();
        break;
      case "D":
        if (this.waitingIndex >= 0) {
          this.clearWaiting();
          this.resetReslutList();
        } else if (this.cval.length > 0) {
          this.cval = deleteLast(this.cval);
          this.resetReslutList();
        } else {
          this.$emit("delete", {});
        }
        break;
      case "space":
        this.addAllTxt(" ");
        break;
      case "down":
        this.downFlag = this.downFlag === "down" ? "" : "down";
        break;
      case "select":
        if (this.lastWaitingStr != sign && this.lastWaitingStr) {
          if (this.lang === "cn") {
            this.cval += this.waitingList[this.waitingIndex];
          } else {
            this.addAllTxt(this.waitingList[this.waitingIndex].toLowerCase());
          }
          this.clearWaiting();
          this.resetReslutList();
        }
        break;
      case "switchNum":
        this.numFlag = false;
        this.numFlag_jp = false;
        this.cval = "";
        this.clearWaiting();
        this.resetReslutList();
        break;
      case "switchNum_jp":
        this.numFlag = false;
        this.numFlag_jp = false;
        this.cval = "";
        this.clearWaiting();
        this.resetReslutList();
        break;
      case "switchCn":
        this.numFlag = false;
        this.numFlag_jp = false;
        break;
      case "switchUpper":
        this.upperFlag = false;
        break;
      case "switchLow":
        this.upperFlag = false;
        break;
      default:
        if (sign.length === 1) {
          this.addAllTxt(sign);
        } else {
          if (this.waitingIndex >= 0) {
            if (this.lastWaitingStr === sign) {
              this.waitingIndex++;
              if (this.waitingIndex >= this.lastWaitingStr.length) {
                this.waitingIndex = 0;
              }
            } else {
              if (this.lang === "cn") {
                this.cval += this.waitingList[this.waitingIndex];
              } else {
                this.addAllTxt(
                  this.waitingList[this.waitingIndex].toLowerCase(),
                );
              }
              this.lastWaitingStr = sign;
              this.waitingIndex = 0;
              this.waitingList = sign.split("");
            }
          } else {
            this.lastWaitingStr = sign;
            this.waitingIndex = 0;
            this.waitingList = sign.split("");
          }
          this.resetReslutList();
        }
        break;
    }
  },
  clearWaiting() {
    this.waitingList = [];
    this.waitingIndex = -1;
    this.lastWaitingStr = "";
  },
  resetReslutList() {
    let watingStr = "";
    if (this.lastWaitingStr && this.lastWaitingStr[this.waitingIndex]) {
      watingStr = this.lastWaitingStr[this.waitingIndex];
    }
    if (this.lang === "en") {
      this.getEnglishResults(this.englishseed);
      return;
    }
    if (!(this.cval + watingStr) || this.lang !== "cn") {
      this.resultList = [];
      this.setResultListAll();
      return;
    }
    this.getResultByWord(this.cval + watingStr);
  },
  setResultListAll() {
    this.resultList2 = [];
    let array = [];
    const rowSize = this.lang === "en" ? 1 : parseInt(this.maxlength);
    for (let i = 0; i < this.resultList.length; i++) {
      array.push(this.resultList[i]);
      if (array.length === rowSize) {
        this.resultList2.push(array);
        array = [];
      }
    }
    if (array.length > 0 && array.length < rowSize) {
      this.resultList2.push(array);
    }
  },
  getResultByWord(val) {
    const that = this;
    doSearchDic(val, that.lang, function (data) {
      that.resultList = data;
      that.setResultListAll();
    });
  },
  getEnglishResults(seedValue) {
    const seed = this.normalizeEnglishSeed(seedValue);
    if (!seed) {
      this.resultList = [];
      this.setResultListAll();
      return;
    }
    this.resultList = this.parseEnglishSuggestions(this.englishsuggestions);
    this.setResultListAll();
  },
  onSelect(num) {
    this.$emit("keyDown", { content: num });
    if (this.keyboardtype === "T9" && this.screentype !== "pill-shaped") {
      this.onBtnClick(num);
      return;
    }
    this.onVibrate();
    if (this.lang === "cn" && !this.numFlag) {
      this.cval += num.toLowerCase();
    } else if (this.lang === "en" && !this.numFlag) {
      this.addAllTxt(num.toLowerCase());
    } else {
      this.addAllTxt(num);
    }
    this.resetReslutList();
  },
  onSelectWaiting(num) {
    this.onVibrate();
    if (this.lang === "cn") {
      this.cval += this.waitingList[num].toString();
    } else {
      this.addAllTxt(this.waitingList[num].toLowerCase());
    }
    this.clearWaiting();
    this.resetReslutList();
  },
  watchHidePropsChange(newV) {
    this.$emit("visibilityChange", { visible: !newV });
  },
  watchMaxLengthPropsChange(newV) {
    if (newV) {
      const tempCvalList = [];
      for (let i = 0; i < newV; i++) {
        tempCvalList.push(i);
      }
      this.cvalList = tempCvalList;
    }
  },
  watchKeyboardTypePropsChange() {
    if (this.lang !== "cn" && this.lang !== "en") {
      this.lang = "cn";
      this.numFlag_jp = false;
      this.cval = "";
      this.clearWaiting();
      this.resetReslutList();
    }
  },
  watchEnglishSeedPropsChange(newV, oldV) {
    if (this.lang === "en" && newV !== oldV) {
      this.resetReslutList();
    }
  },
  watchEnglishSuggestionsPropsChange(newV, oldV) {
    if (this.lang === "en" && newV !== oldV) {
      this.resetReslutList();
    }
  },
  onVibrate() {
    if (this.vibratemode != "") {
      vibrator.vibrate({ mode: this.vibratemode });
    }
  },
  handelScroll(event) {
    let percentTemp67 = (event.scrollX / 636) * 100 + 52.8;
    this.percent67 = parseInt(percentTemp67 <= 100 ? percentTemp67 : 100);
    let percentTemp66 = (event.scrollX / 633) * 100;
    this.percent66 = parseInt(percentTemp66 <= 100 ? percentTemp66 : 100);
  },
  pushCval() {
    this.onVibrate();
    let temp = this.cval;
    this.cval = "";
    this.clearWaiting();
    this.resetReslutList();
    this.addAllTxt(temp);
  },
  normalizeEnglishSeed(value) {
    const text = (value || "").toLowerCase();
    const match = text.match(/[a-z][a-z'-]*$/);
    return match && match[0] ? match[0] : "";
  },
  parseEnglishSuggestions(value) {
    const text = value || "";
    if (!text) {
      return [];
    }
    const parts = text.split("|");
    const output = [];
    for (let i = 0; i < parts.length && output.length < 30; i++) {
      const item = parts[i];
      if (item) {
        output.push(item);
      }
    }
    return output;
  },
  adjustScreenWidth(){
    device.getInfo({
      success: (data) => {
        this.screenWidth = data.screenWidth;
      }
    })
  }
};

import router from "@system.router"

export default {
  private: {
    touchStartX: -1,
    touchStartY: -1
  },

  goBack() {
    router.back()
  },

  onTouchStart(evt) {
    const point = this.getTouchPoint(evt)
    if (point) {
      this.touchStartX = point.clientX
      this.touchStartY = point.clientY
    }
  },

  onTouchEnd(evt) {
    const point = this.getTouchPoint(evt)
    if (!point || this.touchStartX < 0) {
      return
    }

    const endX = point.clientX
    const endY = point.clientY
    const startsInLeftQuarter = this.touchStartX <= 20
    const endsAtRightQuarter = endX >= 180
    const mostlyHorizontal = Math.abs(endY - this.touchStartY) <= 60

    if (startsInLeftQuarter && endsAtRightQuarter && mostlyHorizontal) {
      router.back()
    }

    this.touchStartX = -1
    this.touchStartY = -1
  },

  getTouchPoint(evt) {
    if (evt && evt.changedTouches && evt.changedTouches.length > 0) {
      return evt.changedTouches[0]
    }
    if (evt && evt.touches && evt.touches.length > 0) {
      return evt.touches[0]
    }
    return null
  }
}

<template>
  <div class="page" ontouchstart="onTouchStart" ontouchend="onTouchEnd">
    <div class="back-button" @click="goBack">
      <text class="back-text">返回</text>
    </div>

    <text class="word-text">{{ wordText }}</text>
    <text class="phonetic-text">{{ phoneticText }}</text>

    <list class="meaning-list">
      <list-item class="meaning-item" type="meaningLine" for="{{ line in translationLines }}">
        <text class="meaning-line">{{ line }}</text>
      </list-item>
    </list>

    <div class="bottom-row">
      <input class="bottom-button inflect-button" type="button" value="查变形" @click="goInflect" />
      <input class="bottom-button favorite-button" type="button" value="{{ favoriteLabel }}" @click="toggleFavorite" />
    </div>
  </div>
</template>

<script>
import router from "@system.router"
import storage from "@system.storage"
import file from "@system.file"
import prompt from "@system.prompt"

export default {
  protected: {
    word: "",
    phonetic: "",
    translation: "",
    tag: "",
    searchMode: "",
    inflectDepth: ""
  },

  private: {
    wordText: "",
    phoneticText: "",
    translationText: "",
    tagText: "",
    translationLines: [],
    favoriteLabel: "收藏",
    fileCache: {},
    sourceWord: "",
    inflectForms: [],
    lastToastAt: 0,
    toastCooldown: 1500,
    lastInflectAt: 0,
    inflectPressCooldown: 1000,
    maxInflectDepth: 3,
    touchStartX: -1,
    touchStartY: -1
  },

  onInit() {
    this.wordText = this.word || ""
    this.phoneticText = this.phonetic ? "/" + this.phonetic + "/" : ""
    this.translationText = this.translation || ""
    this.tagText = this.tag || ""
    this.translationLines = this.wrapText(this.translationText, 164)
    if (this.translationLines.length === 0) {
      this.translationLines = [""]
    }
    this.refreshFavoriteState()
  },

  goBack() {
    router.back()
  },

  goInflect() {
    if (!this.wordText) {
      return
    }

    const now = Date.now()
    if (now - this.lastInflectAt < this.inflectPressCooldown) {
      this.showToast("操作太快")
      return
    }
    this.lastInflectAt = now

    const depth = this.currentInflectDepth()
    if (depth >= this.maxInflectDepth) {
      this.showToast("最多查三层")
      return
    }

    router.replace({
      uri: "/pages/results",
      params: {
        query: this.wordText,
        searchMode: "inflect",
        inflectDepth: "" + (depth + 1)
      }
    })
  },

  goInflectParam(queryWord) {
    if (!queryWord) {
      return
    }
    router.push({
      uri: "/pages/results",
      params: {
        query: queryWord,
        searchMode: this.searchMode || "",
        inflectDepth: this.inflectDepth || "0"
      }
    })
  },

  currentInflectDepth() {
    const depth = parseInt(this.inflectDepth)
    if (isNaN(depth) || depth < 0) {
      return 0
    }
    return depth
  },

  showToast(message) {
    const now = Date.now()
    if (now - this.lastToastAt < this.toastCooldown) {
      return
    }
    this.lastToastAt = now
    prompt.showToast({
      message: message,
      duration: 1500
    })
  },

  toggleFavorite() {
    if (!this.wordText) {
      return
    }
    const item = this.currentItem()
    storage.get({
      key: "dic_favorites",
      success: (data) => {
        const list = this.parseList(data)
        this.writeFavoriteList(list, item)
      },
      fail: () => {
        this.writeFavoriteList([], item)
      }
    })
  },

  writeFavoriteList(list, item) {
    const word = this.normalizeWord(item.word)
    let existed = false
    const output = []

    for (let i = 0; i < list.length; i++) {
      const oldItem = list[i]
      if (oldItem && this.normalizeWord(oldItem.word) === word) {
        existed = true
      } else if (oldItem && output.length < 20) {
        output.push(oldItem)
      }
    }

    if (!existed) {
      output.unshift(item)
      while (output.length > 20) {
        output.pop()
      }
      this.favoriteLabel = "已收藏"
    } else {
      this.favoriteLabel = "收藏"
    }

    storage.set({
      key: "dic_favorites",
      value: JSON.stringify(output)
    })
  },

  refreshFavoriteState() {
    storage.get({
      key: "dic_favorites",
      success: (data) => {
        const list = this.parseList(data)
        const word = this.normalizeWord(this.wordText)
        this.favoriteLabel = "收藏"
        for (let i = 0; i < list.length; i++) {
          if (list[i] && this.normalizeWord(list[i].word) === word) {
            this.favoriteLabel = "已收藏"
            return
          }
        }
      },
      fail: () => {
        this.favoriteLabel = "收藏"
      }
    })
  },

  currentItem() {
    return {
      word: this.wordText,
      phonetic: this.phonetic || "",
      translation: this.translationText,
      tag: this.tagText
    }
  },

  parseList(data) {
    if (!data) {
      return []
    }
    const text = data.value ? data.value : data
    try {
      const list = JSON.parse(text)
      if (list && list.length) {
        return list
      }
    } catch (err) {
      return []
    }
    return []
  },

  wrapText(value, maxWidth) {
    const text = (value || "").replace(/\s+/g, " ").trim()
    if (!text) {
      return []
    }

    const lines = []
    let line = ""
    let width = 0
    let lastBreak = -1

    for (let i = 0; i < text.length; i++) {
      const ch = text.charAt(i)
      const chWidth = this.getCharWidth(ch)

      if (width + chWidth > maxWidth && line.length > 0) {
        if (lastBreak > 0) {
          lines.push(line.slice(0, lastBreak).trim())
          line = line.slice(lastBreak).trim() + ch
        } else {
          lines.push(line.trim())
          line = ch
        }
        width = this.measureTextWidth(line)
        lastBreak = this.findLastBreak(line)
      } else {
        line += ch
        width += chWidth
      }

      if (ch === " " || ch === ";" || ch === "," || ch === "，" || ch === "；") {
        lastBreak = line.length
      }
    }

    if (line.trim()) {
      lines.push(line.trim())
    }
    return lines
  },

  measureTextWidth(value) {
    let width = 0
    for (let i = 0; i < value.length; i++) {
      width += this.getCharWidth(value.charAt(i))
    }
    return width
  },

  findLastBreak(value) {
    for (let i = value.length - 1; i >= 0; i--) {
      const ch = value.charAt(i)
      if (ch === " " || ch === ";" || ch === "," || ch === "，" || ch === "；") {
        return i + 1
      }
    }
    return -1
  },

  getCharWidth(ch) {
    if (!ch) {
      return 0
    }
    if (ch >= "\u4e00" && ch <= "\u9fff") {
      return 15
    }
    if (ch >= "A" && ch <= "Z") {
      return 9
    }
    if (ch >= "a" && ch <= "z") {
      return 8
    }
    if (ch >= "0" && ch <= "9") {
      return 8
    }
    if (ch === " " || ch === "." || ch === "," || ch === ";" || ch === ":" || ch === "/" || ch === "-") {
      return 5
    }
    return 9
  },

  normalizeWord(value) {
    return (value || "").toLowerCase().trim()
  },

  keyFor(value) {
    const word = this.normalizeWord(value)
    let key = ""
    for (let i = 0; i < word.length && key.length < 2; i++) {
      const ch = word.charAt(i)
      if ((ch >= "a" && ch <= "z") || (ch >= "0" && ch <= "9")) {
        key += ch
      } else {
        key += "_"
      }
    }
    while (key.length < 2) {
      key += "_"
    }
    return key
  },

  readText(uri, done) {
    if (this.fileCache[uri]) {
      done(this.fileCache[uri])
      return
    }
    file.readText({
      uri: uri,
      encoding: "utf-8",
      success: (data) => {
        const text = data && data.text ? data.text : ""
        this.fileCache[uri] = text
        done(text)
      },
      fail: () => {
        this.fileCache[uri] = ""
        done("")
      }
    })
  },

  scanForwardInflect(text, word) {
    const lines = text.split("\n")
    const lowerWord = this.normalizeWord(word)
    for (let i = 0; i < lines.length; i++) {
      const parts = lines[i].split("\t")
      if (parts.length >= 2 && this.normalizeWord(parts[0]) === lowerWord) {
        return parts[1].split(",").map((s) => s.trim()).filter((s) => s)
      }
    }
    return []
  },

  scanReverseInflect(text, word) {
    const lines = text.split("\n")
    const lowerWord = this.normalizeWord(word)
    for (let i = 0; i < lines.length; i++) {
      const parts = lines[i].split("\t")
      if (parts.length >= 2 && this.normalizeWord(parts[0]) === lowerWord) {
        return parts[1].split(",").map((s) => s.trim()).filter((s) => s)
      }
    }
    return []
  },

  loadInflectForms() {
    const word = this.normalizeWord(this.wordText)
    if (!word || word.length < 2) {
      return
    }

    const shard = this.keyFor(word)
    const self = this

    this.readText("/common/dict/inflect/inflect_" + shard + ".txt", (text) => {
      const bases = text ? self.scanForwardInflect(text, word) : []
      if (bases.length > 0) {
        self.loadReverseInflectForms(bases[0], word, true)
        return
      }
      self.loadReverseInflectForms(word, word, false)
    })
  },

  loadReverseInflectForms(base, currentWord, showSource) {
    const revShard = this.keyFor(base)
    this.readText("/common/dict/inflect_reverse/ireverse_" + revShard + ".txt", (revText) => {
      if (!revText) {
        return
      }
      const allForms = this.scanReverseInflect(revText, base)
      if (allForms.length > 0) {
        if (showSource) {
          this.sourceWord = base
        }
        this.inflectForms = allForms.filter((f) => f !== currentWord).slice(0, 10)
      }
    })
  },

  onTouchStart(evt) {
    const point = this.getTouchPoint(evt)
    if (point) {
      this.touchStartX = point.clientX
      this.touchStartY = point.clientY
    }
  },

  onTouchEnd(evt) {
    const point = this.getTouchPoint(evt)
    if (!point || this.touchStartX < 0) {
      return
    }

    const endX = point.clientX
    const endY = point.clientY
    const startsInLeftQuarter = this.touchStartX <= 53
    const endsAtRightQuarter = endX >= 159
    const mostlyHorizontal = Math.abs(endY - this.touchStartY) <= 120

    if (startsInLeftQuarter && endsAtRightQuarter && mostlyHorizontal) {
      router.back()
    }

    this.touchStartX = -1
    this.touchStartY = -1
  },

  getTouchPoint(evt) {
    if (evt && evt.changedTouches && evt.changedTouches.length > 0) {
      return evt.changedTouches[0]
    }
    if (evt && evt.touches && evt.touches.length > 0) {
      return evt.touches[0]
    }
    return null
  }
}
</script>

<script>
import router from "@system.router"
import storage from "@system.storage"

export default {
  protected: {
    type: ""
  },

  private: {
    records: [],
    emptyText: "",
    touchStartX: -1,
    touchStartY: -1
  },

  onInit() {
    this.loadRecords()
  },

  onShow() {
    this.loadRecords()
  },

  goBack() {
    router.back()
  },

  loadRecords() {
    const key = this.getStorageKey()
    this.emptyText = this.type === "favorites" ? "暂无收藏" : "暂无历史"
    storage.get({
      key: key,
      success: (data) => {
        this.records = this.buildRecords(this.parseList(data))
      },
      fail: () => {
        this.records = []
      }
    })
  },

  openRecord(item) {
    if (!item) {
      return
    }
    if (this.type === "history") {
      router.replace({
        uri: "/pages/search",
        params: {
          queryParam: item.query || "",
          autoSearch: "1"
        }
      })
      return
    }
    if (!item.word) {
      return
    }
    router.push({
      uri: "/pages/detail",
      params: {
        word: item.word,
        phonetic: item.phonetic || "",
        translation: item.translation || "",
        tag: item.tag || ""
      }
    })
  },

  getStorageKey() {
    if (this.type === "favorites") {
      return "dic_favorites"
    }
    return "dic_history"
  },

  buildRecords(list) {
    const output = []
    for (let i = 0; i < list.length && output.length < 20; i++) {
      const item = list[i]
      if (!item) {
        continue
      }
      if (this.type === "history") {
        if (!item.query) {
          continue
        }
        output.push({
          id: this.normalizeWord(item.query) + "_" + i,
          title: item.query,
          query: item.query,
          shortTranslation: "搜索词",
          metaText: "" + (i + 1)
        })
      } else {
        if (!item.word) {
          continue
        }
        output.push({
          id: this.normalizeWord(item.word) + "_" + i,
          title: item.word,
          word: item.word,
          phonetic: item.phonetic || "",
          translation: item.translation || "",
          tag: item.tag || "",
          shortTranslation: this.clipText(item.translation || "", 16),
          metaText: this.buildMetaText(item.tag || "", i)
        })
      }
    }
    return output
  },

  buildMetaText(tag, index) {
    if (tag) {
      return this.clipText(tag, 5)
    }
    return "" + (index + 1)
  },

  clipText(value, maxLen) {
    const text = (value || "").replace(/\s+/g, " ").trim()
    if (text.length <= maxLen) {
      return text
    }
    return text.slice(0, maxLen) + "..."
  },

  parseList(data) {
    if (!data) {
      return []
    }
    const text = data.value ? data.value : data
    try {
      const list = JSON.parse(text)
      if (list && list.length) {
        return list
      }
    } catch (err) {
      return []
    }
    return []
  },

  normalizeWord(value) {
    return (value || "").toLowerCase().trim()
  },

  onTouchStart(evt) {
    const point = this.getTouchPoint(evt)
    if (point) {
      this.touchStartX = point.clientX
      this.touchStartY = point.clientY
    }
  },

  onTouchEnd(evt) {
    const point = this.getTouchPoint(evt)
    if (!point || this.touchStartX < 0) {
      return
    }

    const endX = point.clientX
    const endY = point.clientY
    const startsInLeftQuarter = this.touchStartX <= 53
    const endsAtRightQuarter = endX >= 159
    const mostlyHorizontal = Math.abs(endY - this.touchStartY) <= 120

    if (startsInLeftQuarter && endsAtRightQuarter && mostlyHorizontal) {
      router.back()
    }

    this.touchStartX = -1
    this.touchStartY = -1
  },

  getTouchPoint(evt) {
    if (evt && evt.changedTouches && evt.changedTouches.length > 0) {
      return evt.changedTouches[0]
    }
    if (evt && evt.touches && evt.touches.length > 0) {
      return evt.touches[0]
    }
    return null
  }
}

const FUZZY_SCAN_LIMIT = 4000
const FUZZY_POOL_LIMIT = 80

class LRUCache {
  constructor(max) {
    this.max = max
    this.cache = {}
    this.order = []
  }
  get(key) {
    if (this.cache[key] === undefined) return undefined
    var idx = this.order.indexOf(key)
    if (idx >= 0) {
      this.order.splice(idx, 1)
      this.order.push(key)
    }
    return this.cache[key]
  }
  set(key, val) {
    if (this.cache[key] !== undefined) {
      var idx = this.order.indexOf(key)
      if (idx >= 0) this.order.splice(idx, 1)
    }
    this.cache[key] = val
    this.order.push(key)
    if (this.order.length > this.max) {
      var oldest = this.order.shift()
      delete this.cache[oldest]
    }
  }
}

export default {
  protected: {

    searchMode: "",
    inflectDepth: ""
  },

  private: {

  onInit() {

  performSearch() {
    const normalized = this.normalizeWord(this.queryText)

    if (!normalized) {

    }

    const token = this.searchToken + 1
    this.searchToken = token

    if (this.hasChinese(normalized)) {

      }

      this.collectShardResults(shards, normalized, results, seen, 0, () => {
        this.collectInflectResult(normalized, results, seen, () => {

          }

  finishEnglishSearch(query, token, results) {
    if (results.length === 0 && this.searchMode !== "inflect" && this.isEnglishWord(query)) {
      this.showToast("未找到，混淆搜索")
      this.statusText = "深度搜索"

  performChineseSearch(query, token) {
    const normalizedQuery = this.normalizeChineseText(query)
    if (!normalizedQuery) {
      this.resultCards = []
      this.statusText = ""

    }

    this.loadChineseIds(normalizedQuery, (ids) => {
      if (token !== this.searchToken) {

      }

      this.collectChineseResults(ids, normalizedQuery, results, seen, 0, () => {

        if (this.resultCards.length === 0) {
          this.statusText = "未找到"
        } else if (results.length >= 20) {
          this.statusText = "最多显示20条"

  loadChineseIds(query, done) {
    const firstChar = this.getFirstChineseChar(query)
    if (!firstChar) {
      done([])

    const bucket = this.zhBucketFor(firstChar)
    this.readText("/common/dict/zh_index/zh_" + bucket + ".txt", (text) => {

  collectChineseResults(ids, query, results, seen, index, done) {

        if (nextId < 0 || this.entryShardFor(nextId) !== shard) {

        if (item) {

  loadCnIndex(bucket, done) {
    const cached = this.cnIndexCache.get(bucket)
    if (cached) {

    this.readText("/common/dict/cn_index/cn_" + bucket + ".txt", (text) => {

      const lines = text.split("\n")
      for (let i = 0; i < lines.length; i++) {

        if (parts.length >= 2) {

      this.cnIndexCache.set(bucket, indexMap)

  collectCn2EnResults(query, results, seen, done) {
    const firstChar = this.getFirstChineseChar(query)
    if (!firstChar) {
      done(results)
      return
    }

    const bucket = this.zhBucketFor(firstChar)
    this.loadCnIndex(bucket, (indexMap) => {
      let ids = indexMap[query]

      // Prefix match fallback: max 20 candidate keys, 10 IDs per key, 50 total
      if (!ids) {
        const candidates = Object.keys(indexMap)
          .filter(key => key.indexOf(query) === 0)
          .slice(0, 20)

        const allIds = []
        const idSeen = {}
        for (let i = 0; i < candidates.length; i++) {
          const idList = indexMap[candidates[i]] || []
          const limited = idList.slice(0, 10)
          for (let j = 0; j < limited.length; j++) {
            if (!idSeen[limited[j]]) {
              idSeen[limited[j]] = true
              allIds.push(limited[j])
            }
          }

        ids = allIds.slice(0, 50)

      }

      if (!ids || ids.length === 0) {
        done(results)
        return
      }

      this.fetchEntriesByIds(ids, results, seen, 0, done)
    })

  fetchEntriesByIds(ids, results, seen, index, done) {
    if (index >= ids.length || results.length >= 20) {

    const entryId = this.parseEntryId(ids[index])
    if (entryId < 0) {

      while (nextIndex < ids.length && results.length < 20) {

  loadShardIndex(done) {

        const line = lines[i].trim()
        if (!line) {

        }
        const parts = line.split("\t")
        if (parts.length >= 2) {
          index[parts[0]] = parts[1].split(",")
        }

      done()
    })

  getCandidateShards(query) {
    if (query.length === 1 && this.shardIndex && this.shardIndex[query]) {
      return this.shardIndex[query]
    }
    return [this.keyFor(query)]
  },

  collectShardResults(shards, query, results, seen, index, done) {
    if (index >= shards.length || results.length >= 20) {
      done()
      return
    }

    const shard = shards[index]
    this.readText("/common/dict/words/dict_" + shard + ".txt", (text) => {
      this.scanWordText(text, query, results, seen, "")
      this.collectShardResults(shards, query, results, seen, index + 1, done)
    })

  collectInflectResult(query, results, seen, done) {
    if (results.length >= 20 || query.length < 2) {
      done()
      return
    }

    const shard = this.keyFor(query)
    this.readText("/common/dict/inflect/inflect_" + shard + ".txt", (text) => {
      const bases = []
      const lines = text.split("\n")
      for (let i = 0; i < lines.length; i++) {
        const parts = lines[i].split("\t")
        if (parts.length >= 2 && parts[0] === query) {
          const words = parts[1].split(",")
          for (let j = 0; j < words.length; j++) {
            if (words[j]) {
              bases.push(words[j])
            }
          }
          break
        }
      }
      this.collectBaseResults(bases, results, seen, 0, done)
    })
  },

  collectBaseResults(bases, results, seen, index, done) {
    if (index >= bases.length || results.length >= 20) {
      done()
      return
    }

    const base = bases[index]
    const shard = this.keyFor(base)
    this.readText("/common/dict/words/dict_" + shard + ".txt", (text) => {
      this.scanWordText(text, this.normalizeWord(base), results, seen, "变形")
      this.collectBaseResults(bases, results, seen, index + 1, done)
    })
  },

  scanWordText(text, query, results, seen, source) {
    if (!text) {
      return
    }
    const lines = text.split("\n")
    for (let i = 0; i < lines.length && results.length < 20; i++) {
      const parts = lines[i].split("\t")
      if (parts.length < 3) {
        continue
      }

      const word = parts[0]
      const lower = this.normalizeWord(word)
      if (!lower || seen[lower]) {
        continue
      }
      if (source) {
        if (lower !== query) {
          continue
        }
      } else if (lower.indexOf(query) !== 0) {
        continue
      }

      seen[lower] = true
      results.push(this.createResultCard(lower, word, parts[1], parts[2], source || parts[3] || ""))
    }
  },

  collectFuzzyResults(query, done) {
    const shards = this.getFuzzyShards(query)
    const items = []
    const seen = {}
    this.fuzzyWordsChecked = 0
    this.collectFuzzyShardResults(shards, query, items, seen, 0, () => {
      items.sort((left, right) => {
        if (left.distance !== right.distance) {
          return left.distance - right.distance
        }
        if (right.score !== left.score) {
          return right.score - left.score
        }
        return left.word.length - right.word.length
      })

      const cards = []
      for (let i = 0; i < items.length && cards.length < 20; i++) {
        const item = items[i]
        cards.push(
          this.createResultCard(item.lower, item.word, item.phonetic, item.translation, "相近")
        )
      }
      done(cards)
    })
  },

  getFuzzyShards(query) {
    const ordered = []
    const seen = {}
    const first = query.charAt(0)
    if (this.shardIndex && this.shardIndex[first]) {
      this.pushUniqueShards(ordered, seen, this.shardIndex[first])
    }
    if (this.shardIndex) {
      for (const key in this.shardIndex) {
        this.pushUniqueShards(ordered, seen, this.shardIndex[key])
      }
    }
    return ordered
  },

  pushUniqueShards(output, seen, shards) {
    for (let i = 0; i < shards.length; i++) {
      const shard = shards[i]
      if (shard && !seen[shard]) {
        seen[shard] = true
        output.push(shard)
      }
    }
  },

  collectFuzzyShardResults(shards, query, items, seen, index, done) {
    if (
      index >= shards.length ||
      items.length >= FUZZY_POOL_LIMIT ||
      this.fuzzyWordsChecked >= FUZZY_SCAN_LIMIT
    ) {
      done()
      return
    }

    const shard = shards[index]
    this.readText("/common/dict/words/dict_" + shard + ".txt", (text) => {
      this.scanFuzzyWordText(text, query, items, seen)
      this.collectFuzzyShardResults(shards, query, items, seen, index + 1, done)
    })
  },

  scanFuzzyWordText(text, query, items, seen) {
    if (!text) {
      return
    }
    const lines = text.split("\n")
    for (
      let i = 0;
      i < lines.length &&
      items.length < FUZZY_POOL_LIMIT &&
      this.fuzzyWordsChecked < FUZZY_SCAN_LIMIT;
      i++
    ) {
      this.fuzzyWordsChecked++
      const parts = lines[i].split("\t")
      if (parts.length < 4) {
        continue
      }
      const lower = this.normalizeWord(parts[0])
      if (!lower || seen[lower] || Math.abs(lower.length - query.length) > 2) {
        continue
      }
      const distance = this.editDistanceBounded(query, lower, 2)
      if (distance < 0 || distance > 2) {
        continue
      }
      seen[lower] = true
      items.push({
        lower: lower,
        word: parts[0],
        phonetic: parts[1],
        translation: parts[2],
        distance: distance,
        score: this.scoreFuzzyCandidate(query, lower, parts[3] || "")
      })
    }
  },

  editDistanceBounded(left, right, limit) {
    const leftLength = left.length
    const rightLength = right.length
    if (Math.abs(leftLength - rightLength) > limit) {
      return -1
    }

    let previous = []
    let current = []
    for (let j = 0; j <= rightLength; j++) {
      previous[j] = j
    }

    for (let i = 1; i <= leftLength; i++) {
      current = [i]
      let rowMin = current[0]
      const from = Math.max(1, i - limit)
      const to = Math.min(rightLength, i + limit)

      for (let j = 1; j < from; j++) {
        current[j] = limit + 1
      }
      for (let j = from; j <= to; j++) {
        const cost = left.charAt(i - 1) === right.charAt(j - 1) ? 0 : 1
        const deletion = previous[j] + 1
        const insertion = current[j - 1] + 1
        const substitution = previous[j - 1] + cost
        const value = Math.min(deletion, insertion, substitution)
        current[j] = value
        if (value < rowMin) {
          rowMin = value
        }
      }
      for (let j = to + 1; j <= rightLength; j++) {
        current[j] = limit + 1
      }
      if (rowMin > limit) {
        return -1
      }
      previous = current
    }

    const result = previous[rightLength]
    return result <= limit ? result : -1
  },

  scoreFuzzyCandidate(query, word, tag) {
    let score = 80 - Math.abs(word.length - query.length) * 8
    const tags = tag || ""
    if (word.charAt(0) === query.charAt(0)) {
      score += 20
    }
    if (tags.indexOf("zk") >= 0) {
      score += 40
    }
    if (tags.indexOf("gk") >= 0) {
      score += 36
    }
    if (tags.indexOf("cet4") >= 0) {
      score += 30
    }
    if (tags.indexOf("cet6") >= 0 || tags.indexOf("ky") >= 0) {
      score += 18
    }
    if (tags.indexOf("toefl") >= 0 || tags.indexOf("ielts") >= 0) {
      score += 8
    }
    if (tags.indexOf("gre") >= 0) {
      score -= 8
    }
    return score
  },

  createResultCard(id, word, phonetic, translation, tag) {
    const lines = this.wrapText(translation || "", 168)
    const safeLines = lines.length > 0 ? lines : [""]
    const lineHeight = 22
    const cardHeight = Math.max(100, 64 + safeLines.length * lineHeight)

    return {
      id: id,
      word: word,
      phonetic: phonetic ? "/" + phonetic + "/" : "",
      rawPhonetic: phonetic || "",
      translation: translation || "",
      translationLines: safeLines,
      tag: tag,
      cardHeight: cardHeight,
      itemHeight: cardHeight + 10
    }
  },

  wrapText(value, maxWidth) {
    const text = (value || "").replace(/\s+/g, " ").trim()
    if (!text) {
      return []
    }

    const lines = []
    let line = ""
    let width = 0
    let lastBreak = -1

    for (let i = 0; i < text.length; i++) {
      const ch = text.charAt(i)
      const chWidth = this.getCharWidth(ch)

      if (width + chWidth > maxWidth && line.length > 0) {
        if (lastBreak > 0) {
          lines.push(line.slice(0, lastBreak).trim())
          line = line.slice(lastBreak).trim() + ch
        } else {
          lines.push(line.trim())
          line = ch
        }
        width = this.measureTextWidth(line)
        lastBreak = this.findLastBreak(line)
      } else {
        line += ch
        width += chWidth
      }

      if (ch === " " || ch === ";" || ch === "," || ch === "，" || ch === "；") {
        lastBreak = line.length
      }
    }

    if (line.trim()) {
      lines.push(line.trim())
    }
    return lines
  },

  measureTextWidth(value) {
    let width = 0
    for (let i = 0; i < value.length; i++) {
      width += this.getCharWidth(value.charAt(i))
    }
    return width
  },

  findLastBreak(value) {
    for (let i = value.length - 1; i >= 0; i--) {
      const ch = value.charAt(i)
      if (ch === " " || ch === ";" || ch === "," || ch === "，" || ch === "；") {
        return i + 1
      }
    }
    return -1
  },

  getCharWidth(ch) {
    if (!ch) {
      return 0
    }
    if (ch >= "\u4e00" && ch <= "\u9fff") {
      return 13
    }
    if (ch >= "A" && ch <= "Z") {
      return 8
    }
    if (ch >= "a" && ch <= "z") {
      return 7
    }
    if (ch >= "0" && ch <= "9") {
      return 7
    }
    if (
      ch === " " ||
      ch === "." ||
      ch === "," ||
      ch === ";" ||
      ch === ":" ||
      ch === "/" ||
      ch === "-"
    ) {
      return 4
    }
    return 8
  },

  readText(uri, done) {
    const cached = this.fileCache.get(uri)
    if (cached !== undefined) {
      done(cached)
      return
    }

    file.readText({
      uri: uri,
      encoding: "utf-8",
      success: (data) => {
        const text = data && data.text ? data.text : ""
        this.fileCache.set(uri, text)
        done(text)
      },
      fail: () => {
        this.fileCache.set(uri, "")
        done("")
      }
    })
  },

  normalizeWord(value) {
    return (value || "").toLowerCase().trim()
  },

  isEnglishWord(value) {
    return /^[a-z][a-z'-]*$/.test(value || "")
  },

  showToast(message) {
    const now = Date.now()
    if (now - this.lastToastAt < this.toastCooldown) {
      return
    }
    this.lastToastAt = now
    prompt.showToast({
      message: message,
      duration: 1500
    })
  },

  hasChinese(value) {
    const text = value || ""
    for (let i = 0; i < text.length; i++) {
      if (this.isChineseChar(text.charAt(i))) {
        return true
      }
    }
    return false
  },

  normalizeChineseText(value) {
    const text = value || ""
    let output = ""
    for (let i = 0; i < text.length; i++) {
      const ch = text.charAt(i)
      if (this.isChineseChar(ch)) {
        output += ch
      }
    }
    return output
  },

  getFirstChineseChar(value) {
    const text = value || ""
    for (let i = 0; i < text.length; i++) {
      const ch = text.charAt(i)
      if (this.isChineseChar(ch)) {
        return ch
      }
    }
    return ""
  },

  isChineseChar(ch) {
    return ch >= "\u4e00" && ch <= "\u9fff"
  },

  zhBucketFor(ch) {
    return this.padHex(ch.charCodeAt(0) % 64)
  },

  padHex(value) {
    const hex = value.toString(16)
    if (hex.length < 2) {
      return "0" + hex
    }
    return hex
  },

  parseEntryId(value) {
    const id = parseInt(value)
    if (isNaN(id)) {
      return -1
    }
    return id
  },

  entryShardFor(entryId) {
    const index = Math.floor(entryId / 500)
    if (index < 10) {
      return "0" + index
    }
    return "" + index
  },

  keyFor(value) {
    const word = this.normalizeWord(value)
    let key = ""
    for (let i = 0; i < word.length && key.length < 2; i++) {
      const ch = word.charAt(i)
      if ((ch >= "a" && ch <= "z") || (ch >= "0" && ch <= "9")) {
        key += ch
      } else {
        key += "_"
      }
    }
    while (key.length < 2) {
      key += "_"
    }
    return key
  },

  onTouchStart(evt) {
    const point = this.getTouchPoint(evt)
    if (point) {
      this.touchStartX = point.clientX
      this.touchStartY = point.clientY
    }
  },

  onTouchEnd(evt) {
    const point = this.getTouchPoint(evt)
    if (!point || this.touchStartX < 0) {
      return
    }

    const endX = point.clientX
    const endY = point.clientY
    const startsInLeftQuarter = this.touchStartX <= 53
    const endsAtRightQuarter = endX >= 159
    const mostlyHorizontal = Math.abs(endY - this.touchStartY) <= 120

    if (startsInLeftQuarter && endsAtRightQuarter && mostlyHorizontal) {
      router.back()
    }

    this.touchStartX = -1
    this.touchStartY = -1
  },

  getTouchPoint(evt) {
    if (evt && evt.changedTouches && evt.changedTouches.length > 0) {
      return evt.changedTouches[0]
    }
    if (evt && evt.touches && evt.touches.length > 0) {
      return evt.touches[0]
    }
    return null
  }
}
</script>

// ENGLISH_SUGGESTION_BUCKETS (327KB) is loaded as a JSON resource via file.readText()
// to avoid bloating the search.js bundle
let _suggestionBuckets = null

function readResourceJSON(uri) {
  return new Promise((resolve, reject) => {
    file.readText({
      uri: uri,
      encoding: "utf-8",
      success: (data) => {
        try {
          resolve(JSON.parse(data.text))
        } catch (e) {
          reject(e)
        }
      },
      fail: (data, code) => {
        reject(new Error("file.readText failed: " + code))
      }
    })
  })
}

async function ensureSuggestionBuckets() {
  if (!_suggestionBuckets) {
    _suggestionBuckets = await readResourceJSON("/Common/english_suggestions.json")
  }
  return _suggestionBuckets
}

const ENGLISH_SUFFIXES = [
  "s",
  "ed",
  "ing",
  "er",
  "est",
  "ly",
  "tion",
  "ment",
  "ness",
  "able",
  "ful",
  "less",
  "ive",
  "al",
  "ous",
  "ity"

export default {
  protected: {
    queryParam: "",
    autoSearch: "",
    searchMode: "",
    inflectDepth: ""
  },

  private: {
    query: "",
    cursorIndex: 0,
    displayText: "",
    englishSeed: "",
    englishSuggestionsText: "",
    keyboardHidden: false,
    keyboardType: "QWERTY",
    vibrateMode: "short",
    screenType: "pill-shaped",
    lastToastAt: 0,
    toastCooldown: 1500,
    touchStartX: -1,
    touchStartY: -1
  },

  onInit() {
    if (this.queryParam) {
      this.query = this.queryParam
      this.cursorIndex = this.query.length
    }
    this.keyboardHidden = this.autoSearch === "1" ? true : false
    this.refreshDisplay()
    if (this.autoSearch === "1" && this.query) {
      this.performSearch()
    }
  },

  toggleKeyboard() {
    this.keyboardHidden = !this.keyboardHidden
    this.refreshDisplay()
  },

  onVisibilityChange(evt) {
    if (evt && evt.detail) {
      this.keyboardHidden = !evt.detail.visible
    }
    this.refreshDisplay()
  },

  onKeyDown(evt) {
    console.log("key down: " + JSON.stringify(evt))
  },

  onDelete() {
    if (this.cursorIndex > 0) {
      this.query = this.query.slice(0, this.cursorIndex - 1) + this.query.slice(this.cursorIndex)
      this.cursorIndex--

  onComplete(evt) {
    if (evt && evt.detail && evt.detail.content) {
      const content = evt.detail.content
      this.query =
        this.query.slice(0, this.cursorIndex) + content + this.query.slice(this.cursorIndex)
      this.cursorIndex += content.length
    }
    this.refreshDisplay()

  clearText() {

    this.cursorIndex = 0
    this.refreshDisplay()
  },

  moveLeft() {
    if (this.cursorIndex > 0) {
      this.cursorIndex--
    }
    this.refreshDisplay()
  },

  moveRight() {
    if (this.cursorIndex < this.query.length) {
      this.cursorIndex++
    }
    this.refreshDisplay()
  },

  performSearch() {
    const normalized = this.normalizeWord(this.query)
    this.keyboardHidden = true
    this.refreshDisplay()

    if (!normalized) {
      return
    }
    if (!this.validateQuery(normalized)) {
      return
    }

    this.saveSearchHistory(normalized)

    router.push({
      uri: "/pages/results",
      params: {
        query: normalized,
        searchMode: this.searchMode || "",
        inflectDepth: this.inflectDepth || "0"
      }
    })
  },

  validateQuery(query) {
    if (this.searchMode === "inflect") {
      if (!this.isEnglishWord(query)) {
        this.showToast("只能输入英文")
        return false
      }
      return true
    }

    const hasChinese = this.hasChinese(query)
    const hasEnglish = this.hasEnglish(query)
    if (hasChinese && hasEnglish) {
      this.showToast("输入非法")
      return false
    }
    if (hasChinese) {
      if (!this.isChineseQuery(query)) {
        this.showToast("输入非法")
        return false
      }
      return true
    }
    if (!this.isEnglishWord(query)) {
      this.showToast("输入非法")
      return false
    }
    return true
  },

  showToast(message) {
    const now = Date.now()
    if (now - this.lastToastAt < this.toastCooldown) {
      return
    }
    this.lastToastAt = now
    prompt.showToast({
      message: message,
      duration: 1500
    })
  },

  saveSearchHistory(query) {
    const item = {
      query: query
    }
    storage.get({
      key: "dic_history",
      success: (data) => {
        const list = this.parseList(data)
        this.writeHistoryList(list, item)
      },
      fail: () => {
        this.writeHistoryList([], item)
      }
    })
  },

  writeHistoryList(list, item) {
    const output = [item]
    for (let i = 0; i < list.length && output.length < 20; i++) {
      const oldItem = list[i]
      if (oldItem && oldItem.query && this.normalizeWord(oldItem.query) !== item.query) {
        output.push(oldItem)
      }
    }
    storage.set({
      key: "dic_history",
      value: JSON.stringify(output)
    })
  },

  parseList(data) {
    if (!data) {
      return []
    }
    const text = data.value ? data.value : data
    try {
      const list = JSON.parse(text)
      if (list && list.length) {
        return list
      }
    } catch (err) {
      return []
    }
    return []
  },

  onTouchStart(evt) {
    const point = this.getTouchPoint(evt)

  refreshDisplay() {
    if (!this.query) {
      this.cursorIndex = 0
      this.displayText = ""
      this.englishSeed = ""

    }

    }

      (this.keyboardHidden ? "" : "_") +

    this.updateEnglishSeed()
  },

  updateEnglishSeed() {
    const text = this.query.slice(0, this.cursorIndex).toLowerCase()
    const match = text.match(/[a-z][a-z'-]*$/)
    const nextSeed = match && match[0] ? match[0] : ""
    if (nextSeed === this.englishSeed) {
      return
    }
    this.englishSeed = nextSeed
    this.refreshEnglishSuggestions(nextSeed)
  },

  async refreshEnglishSuggestions(seed) {
    const buckets = await ensureSuggestionBuckets()
    const normalized = this.normalizeEnglishSeed(seed)
    if (!normalized) {
      this.englishSuggestionsText = ""
      return
    }

    const source = buckets[normalized.charAt(0)] || ""
    const rows = source ? source.split("\n") : []
    const items = []
    const seen = {}
    this.scanEnglishSuggestionRows(rows, normalized, items, seen)
    this.englishSuggestionsText = this.buildEnglishSuggestions(normalized, items, seen).join("|")
  },

  scanEnglishSuggestionRows(rows, seed, items, seen) {
    for (let i = 0; i < rows.length; i++) {
      const parts = rows[i].split("\t")
      const word = this.normalizeEnglishSeed(parts[0])
      if (!word || word === seed || seen[word] || word.indexOf(seed) !== 0) {
        continue
      }
      seen[word] = true
      items.push({
        word: word,
        score: this.scoreEnglishCandidate(seed, word, parts[1] || "")
      })
    }
  },

  buildEnglishSuggestions(seed, items, seen) {
    items.sort((left, right) => {
      if (right.score !== left.score) {
        return right.score - left.score
      }
      return left.word.length - right.word.length
    })
    const output = []
    for (let i = 0; i < items.length && output.length < 30; i++) {
      output.push(items[i].word)
    }
    if (seed.length >= 2) {
      for (let i = 0; i < ENGLISH_SUFFIXES.length && output.length < 30; i++) {
        const suffix = ENGLISH_SUFFIXES[i]
        const combined = seed + suffix
        if (!seen[combined] && output.indexOf(suffix) < 0) {
          output.push(suffix)
        }
      }
    }
    return output
  },

  scoreEnglishCandidate(seed, word, tag) {
    let score = 100 - (word.length - seed.length) * 4
    const tags = tag || ""
    if (tags.indexOf("zk") >= 0) {
      score += 40
    }
    if (tags.indexOf("gk") >= 0) {
      score += 36
    }
    if (tags.indexOf("cet4") >= 0) {
      score += 30
    }
    if (tags.indexOf("cet6") >= 0 || tags.indexOf("ky") >= 0) {
      score += 18
    }
    if (tags.indexOf("toefl") >= 0 || tags.indexOf("ielts") >= 0) {
      score += 8
    }
    if (tags.indexOf("gre") >= 0) {
      score -= 8
    }
    return score
  },

  normalizeEnglishSeed(value) {
    const text = (value || "").toLowerCase()
    const match = text.match(/[a-z][a-z'-]*$/)
    return match && match[0] ? match[0] : ""
  },

  normalizeWord(value) {
    return (value || "").toLowerCase().trim()
  },

  isEnglishWord(value) {
    return /^[a-z][a-z'-]*$/.test(value || "")
  },

  hasEnglish(value) {
    return /[a-z]/.test(value || "")
  },

  hasChinese(value) {
    const text = value || ""
    for (let i = 0; i < text.length; i++) {
      const ch = text.charAt(i)
      if (ch >= "\u4e00" && ch <= "\u9fff") {
        return true
      }
    }
    return false
  },

  isChineseQuery(value) {
    const text = value || ""
    for (let i = 0; i < text.length; i++) {
      const ch = text.charAt(i)
      if (ch < "\u4e00" || ch > "\u9fff") {
        return false
      }
    }
    return text.length > 0
  }
}

<template>
  <div class="page" ontouchstart="onTouchStart" ontouchend="onTouchEnd">
    <div class="back-button" @click="goBack">
      <text class="back-text">返回</text>
    </div>

    <img class="sponsor-code" src="/common/sponsor-code.png" />
  </div>
</template>

<script>
import router from "@system.router"

export default {
  private: {
    touchStartX: -1,
    touchStartY: -1
  },

  goBack() {
    router.back()
  },

  onTouchStart(evt) {
    const point = this.getTouchPoint(evt)
    if (point) {
      this.touchStartX = point.clientX
      this.touchStartY = point.clientY
    }
  },

  onTouchEnd(evt) {
    const point = this.getTouchPoint(evt)
    if (!point || this.touchStartX < 0) {
      return
    }

    const endX = point.clientX
    const endY = point.clientY
    const startsInLeftQuarter = this.touchStartX <= 53
    const endsAtRightQuarter = endX >= 159
    const mostlyHorizontal = Math.abs(endY - this.touchStartY) <= 120

    if (startsInLeftQuarter && endsAtRightQuarter && mostlyHorizontal) {
      router.back()
    }

    this.touchStartX = -1
    this.touchStartY = -1
  },

  getTouchPoint(evt) {
    if (evt && evt.changedTouches && evt.changedTouches.length > 0) {
      return evt.changedTouches[0]
    }
    if (evt && evt.touches && evt.touches.length > 0) {
      return evt.touches[0]
    }
    return null
  }
}
</script>
