<script>
import device from "@system.device"

export default {
  data: {
    screenShape: "pill-shaped",
    screenWidth: 212,
    screenHeight: 520,
    screenAspectRatio: 0.408,
    screenProfile: "pill-standard"
  },

  updateScreenInfo(data) {
    const appData = this.data
    if (data.screenShape) {
      appData.screenShape = data.screenShape
    }
    if (data.screenWidth) {
      appData.screenWidth = data.screenWidth
    }
    if (data.screenHeight) {
      appData.screenHeight = data.screenHeight
    }
    if (appData.screenHeight > 0) {
      appData.screenAspectRatio =
        Math.round((appData.screenWidth / appData.screenHeight) * 1000) / 1000
    }

    if (appData.screenShape === "circle") {
      appData.screenProfile = "circle"
    } else if (appData.screenShape === "rect") {
      appData.screenProfile = "rect"
    } else if (appData.screenAspectRatio <= 0.4) {
      appData.screenProfile = "pill-compact"
    } else {
      appData.screenProfile = "pill-standard"
    }
  },

  onCreate() {
    console.log("app created")
    device.getInfo({
      success: (data) => {
        this.updateScreenInfo(data)
        const appData = this.data
        console.log(
          "screen: shape=" +
            appData.screenShape +
            " width=" +
            appData.screenWidth +
            " height=" +
            appData.screenHeight +
            " ratio=" +
            appData.screenAspectRatio +
            " profile=" +
            appData.screenProfile
        )
      },
      fail: (data, code) => {
        console.log("device.getInfo failed: " + code)
      }
    })
  },

  onDestroy() {
    console.log("app destroyed")
  }
}
</script>

<script>
import vibrator from "@system.vibrator";
import device from '@system.device'
import { SimpleInputMethod } from "./assets/dicUtil.js";
import {getSuggestionSeed, getSuggestions, onSuggestionsChange} from "../../common/suggestionState"
const KEY_MAPS = {
  full: [
    ["Q", "W", "E", "R", "T", "Y", "U", "I", "O", "P"],
    ["A", "S", "D", "F", "G", "H", "J", "K", "L"],
    ["Z", "X", "C", "V", "B", "N", "M"]
  ],
  sign: [
    ["1", "2", "3", "4", "5", "6", "7", "8", "9", "0"],
    ["~", "!", "@", "#", "%", "\u201c", "\u201d", "*", "?", "/"],
    ["(", ")", "-", "_", ":", ";", "\uff0c", "\u3002", "."]
  ],
  sign_jp: [
    ["1", "2", "3", "4", "5", "6", "7", "8", "9", "0"],
    ["~", "\u2022", "@", "#", "%", "\u300c", "\u300d", "*", "?", "/"],
    ["(", ")", "-", "\u2026", ":", ";", "\u3001", "\u3002", "!"]
  ],
  sign62: [
    ["2", "3", "4", "5", "6", "7", "8", "9"],
    ["!", "@", "#", "%", "\u201c", "\u201d", "*"],
    [")", "-", "_", ":", ";"]
  ],
  sign62_jp: [
    ["2", "3", "4", "5", "6", "7", "8", "9"],
    ["\u2022", "@", "#", "%", "\u300c", "\u300d", "*"],
    [")", "-", "\u2026", ":", ";"]
  ],
  full62: [
    ["W", "E", "R", "T", "Y", "U", "I", "O"],
    ["S", "D", "F", "G", "H", "J", "K"],
    ["X", "C", "V", "B", "N"]
  ],
  t9: [
    ["abc", "def"],
    ["ghi", "jkl", "mno"],
    ["pqrs", "tuv", "wxyz"]
  ]
}
function doSearchDic(word, lang, cb) {
  if (!word) {
    cb([]);
    return;
  }

  const result = SimpleInputMethod.getHanzi(word, lang);
  cb(Array.isArray(result) && result[0] ? result[0] : []);
}

function deleteLast(t) {
  if (t) {
    return t.substr(0, t.length - 1);
  }
  return "";
}
export default {
  props: {
    hide: {
      default: true,
    },
    keyboardtype: {
      default: "QWERTY",
    },
    maxlength: {
      default: 5,
    },
    vibratemode: {
      default: "",
    },
    screentype: {
      default: "circle",
    },
  },
  data: {
    cval: "",
    resultList: [],
    resultList2: [],
    waitingList: [],
    waitingIndex: -1,
    lastWaitingStr: "",
    downFlag: "",
    lang: "cn",
    numFlag: false,
    numFlag_jp: false,
    upperFlag: false,
    cvalList: [0, 1, 2, 3, 4],
    percent67: 52,
    percent66: 0,
    // 针对screenShape为rect的设备，会自动获取screenWidth并绑定到根div
    // 这样便能同时适配n67和o65甚至是后续设备，但实际效果可能受designWidth影响
    screenWidth: 212,
    keyboardLeftOffset: 10, // (212 - 192) / 2, 由 adjustScreenWidth 更新
    keys: KEY_MAPS,
  },
  onInit() {
    if (this.lang !== "cn" && this.lang !== "en") {
      this.lang = "cn";
      this.numFlag_jp = false;
    }
    if (this.maxlength) {
      const tempCvalList = [];
      for (let i = 0; i < this.maxlength; i++) {
        tempCvalList.push(i);
      }
      this.cvalList = tempCvalList;
    }
    if (this.screentype === "rect" || this.screentype === "pill-shaped") {
      this.adjustScreenWidth();
    }
    this.$watch("hide", "watchHidePropsChange");
    this.$watch("maxlength", "watchMaxLengthPropsChange");
    this.$watch("keyboardtype", "watchKeyboardTypePropsChange");
    onSuggestionsChange(function () {
      if (this.lang === "en") {
        this.resetReslutList()
      }
    }.bind(this))
  },
  onDestroy() {
    this.$unwatch("hide")
    this.$unwatch("maxlength")
    this.$unwatch("keyboardtype")
    onSuggestionsChange(null)
  },
  addAllTxt(txt) {
    this.$emit("complete", { content: txt });
  },
  onRsSelect(txt) {
    this.onVibrate();
    let content = txt;
    if (this.lang === "en") {
      const seed = this.normalizeEnglishSeed(getSuggestionSeed());
      const word = this.normalizeEnglishSeed(txt);
      if (seed && word.indexOf(seed) === 0 && word.length > seed.length) {
        content = word.slice(seed.length);
      }
    }
    this.cval = "";
    this.addAllTxt(content);
    this.clearWaiting();
    this.resetReslutList();
    this.downFlag = "";
  },
  onBtnClick(sign) {
    this.onVibrate();
    switch (sign) {
      case "AC":
        this.cval = "";
        this.clearWaiting();
        this.resetReslutList();
        break;
      case "lang":
        this.lang = this.lang === "cn" ? "en" : "cn";
        this.cval = "";
        this.clearWaiting();
        this.resetReslutList();
        break;
      case "D":
        if (this.waitingIndex >= 0) {
          this.clearWaiting();
          this.resetReslutList();
        } else if (this.cval.length > 0) {
          this.cval = deleteLast(this.cval);
          this.resetReslutList();
        } else {
          this.$emit("delete", {});
        }
        break;
      case "space":
        this.addAllTxt(" ");
        break;
      case "down":
        this.downFlag = this.downFlag === "down" ? "" : "down";
        break;
      case "select":
        if (this.lastWaitingStr != sign && this.lastWaitingStr) {
          if (this.lang === "cn") {
            this.cval += this.waitingList[this.waitingIndex];
          } else {
            this.addAllTxt(this.waitingList[this.waitingIndex].toLowerCase());
          }
          this.clearWaiting();
          this.resetReslutList();
        }
        break;
      case "switchNum":
        this.numFlag = false;
        this.numFlag_jp = false;
        this.cval = "";
        this.clearWaiting();
        this.resetReslutList();
        break;
      case "switchNum_jp":
        this.numFlag = false;
        this.numFlag_jp = false;
        this.cval = "";
        this.clearWaiting();
        this.resetReslutList();
        break;
      case "switchCn":
        this.numFlag = false;
        this.numFlag_jp = false;
        break;
      case "switchUpper":
        this.upperFlag = false;
        break;
      case "switchLow":
        this.upperFlag = false;
        break;
      default:
        if (sign.length === 1) {
          this.addAllTxt(sign);
        } else {
          if (this.waitingIndex >= 0) {
            if (this.lastWaitingStr === sign) {
              this.waitingIndex++;
              if (this.waitingIndex >= this.lastWaitingStr.length) {
                this.waitingIndex = 0;
              }
            } else {
              if (this.lang === "cn") {
                this.cval += this.waitingList[this.waitingIndex];
              } else {
                this.addAllTxt(
                  this.waitingList[this.waitingIndex].toLowerCase(),
                );
              }
              this.lastWaitingStr = sign;
              this.waitingIndex = 0;
              this.waitingList = sign.split("");
            }
          } else {
            this.lastWaitingStr = sign;
            this.waitingIndex = 0;
            this.waitingList = sign.split("");
          }
          this.resetReslutList();
        }
        break;
    }
  },
  clearWaiting() {
    this.waitingList = [];
    this.waitingIndex = -1;
    this.lastWaitingStr = "";
  },
  resetReslutList() {
    let watingStr = "";
    if (this.lastWaitingStr && this.lastWaitingStr[this.waitingIndex]) {
      watingStr = this.lastWaitingStr[this.waitingIndex];
    }
    if (this.lang === "en") {
      this.getEnglishResults(getSuggestionSeed());
      return;
    }
    if (!(this.cval + watingStr) || this.lang !== "cn") {
      this.resultList = [];
      this.setResultListAll();
      return;
    }
    this.getResultByWord(this.cval + watingStr);
  },
  setResultListAll() {
    this.resultList2 = [];
    let array = [];
    const rowSize = this.lang === "en" ? 1 : parseInt(this.maxlength);
    for (let i = 0; i < this.resultList.length; i++) {
      array.push(this.resultList[i]);
      if (array.length === rowSize) {
        this.resultList2.push(array);
        array = [];
      }
    }
    if (array.length > 0 && array.length < rowSize) {
      this.resultList2.push(array);
    }
  },
  getResultByWord(val) {
    const that = this;
    doSearchDic(val, that.lang, function (data) {
      that.resultList = data;
      that.setResultListAll();
    });
  },
  getEnglishResults(seedValue) {
    const seed = this.normalizeEnglishSeed(seedValue);
    if (!seed) {
      this.resultList = [];
      this.setResultListAll();
      return;
    }
    this.resultList = this.parseEnglishSuggestions(getSuggestions());
    this.setResultListAll();
  },
  onSelect(num) {
    this.$emit("keyDown", { content: num });
    if (this.keyboardtype === "T9" && this.screentype !== "pill-shaped") {
      this.onBtnClick(num);
      return;
    }
    this.onVibrate();
    if (this.lang === "cn" && !this.numFlag) {
      this.cval += num.toLowerCase();
    } else if (this.lang === "en" && !this.numFlag) {
      this.addAllTxt(num.toLowerCase());
    } else {
      this.addAllTxt(num);
    }
    this.resetReslutList();
  },
  onSelectWaiting(num) {
    this.onVibrate();
    if (this.lang === "cn") {
      this.cval += this.waitingList[num].toString();
    } else {
      this.addAllTxt(this.waitingList[num].toLowerCase());
    }
    this.clearWaiting();
    this.resetReslutList();
  },
  watchHidePropsChange(newV) {
    this.$emit("visibilityChange", { visible: !newV });
  },
  watchMaxLengthPropsChange(newV) {
    if (newV) {
      const tempCvalList = [];
      for (let i = 0; i < newV; i++) {
        tempCvalList.push(i);
      }
      this.cvalList = tempCvalList;
    }
  },
  watchKeyboardTypePropsChange() {
    if (this.lang !== "cn" && this.lang !== "en") {
      this.lang = "cn";
      this.numFlag_jp = false;
      this.cval = "";
      this.clearWaiting();
      this.resetReslutList();
    }
  },
  onVibrate() {
    if (this.vibratemode != "") {
      vibrator.vibrate({ mode: this.vibratemode });
    }
  },
  handelScroll(event) {
    let percentTemp67 = (event.scrollX / 636) * 100 + 52.8;
    this.percent67 = parseInt(percentTemp67 <= 100 ? percentTemp67 : 100);
    let percentTemp66 = (event.scrollX / 633) * 100;
    this.percent66 = parseInt(percentTemp66 <= 100 ? percentTemp66 : 100);
  },
  pushCval() {
    this.onVibrate();
    let temp = this.cval;
    this.cval = "";
    this.clearWaiting();
    this.resetReslutList();
    this.addAllTxt(temp);
  },
  normalizeEnglishSeed(value) {
    const text = (value || "").toLowerCase();
    const match = text.match(/[a-z][a-z'-]*$/);
    return match && match[0] ? match[0] : "";
  },
  parseEnglishSuggestions(value) {
    const text = value || "";
    if (!text) {
      return [];
    }
    const parts = text.split("|");
    const output = [];
    for (let i = 0; i < parts.length && output.length < 30; i++) {
      const item = parts[i];
      if (item) {
        output.push(item);
      }
    }
    return output;
  },
  adjustScreenWidth(){
    device.getInfo({
      success: (data) => {
        this.screenWidth = data.screenWidth;
        this.keyboardLeftOffset = Math.max(0, (data.screenWidth - 192) / 2);
        this.$emit("ready", {});
      },
      fail: () => {
        this.keyboardLeftOffset = Math.max(0, (this.screenWidth - 192) / 2);
        this.$emit("ready", {});
      }
    })
  }
};
</script>

<template>
  <div class="page" ontouchstart="onTouchStart" ontouchend="onTouchEnd">
    <div class="back-button" @click="goBack">
      <text class="back-text">返回</text>
    </div>

        scrollamount="{{ wordMarqueeSpeed }}" direction="left" loop="-1" text-offset="40">
        {{ wordText }}   {{ wordText }}
      </marquee>
      <text if="{{ !wordMarqueeOn }}" class="word-text" style="width: {{contentWidth}}px;">{{ wordText }}</text>
      <text class="phonetic-text" style="width: {{contentWidth}}px;">{{ phoneticText }}</text>
    </div>

<script>
import router from "@system.router"
import storage from "@system.storage"
import file from "@system.file"
import prompt from "@system.prompt"

export default {
  protected: {
    word: "",
    phonetic: "",
    translation: "",
    tag: "",
    searchMode: "",
    inflectDepth: ""
  },

  private: {
    wordText: "",
    phoneticText: "",
    translationText: "",
    tagText: "",
    tagLines: [],
    translationLines: [],
    cardWidth: 188,
    contentWidth: 164,
    favoriteLabel: "收藏",
    fileCache: {},
    sourceWord: "",
    inflectForms: [],
    lastToastAt: 0,
    toastCooldown: 1500,
    lastInflectAt: 0,
    inflectPressCooldown: 1000,
    maxInflectDepth: 3,
    touchStartX: -1,
    touchStartY: -1,
    destroyed: false,
    wordMarqueeOn: false,
    wordMarqueeSpeed: 40,
    wordMarqueeThreshold: 9
  },

  onInit() {
    const sw = this.$app.$def.data.screenWidth || 212
    this.cardWidth = sw - 20
    this.contentWidth = this.cardWidth - 24
    this.isRoundScreen = sw === 466
    this.isCapsule = sw === 192 || sw === 212
    if (this.isRoundScreen) {
      this.bottomStyle = "justify-content: center;"
      this.inflectBtnStyle = "margin-right: 5px;"
      this.favoriteBtnStyle = "margin-left: 5px;"
    } else if (this.isCapsule) {
      this.bottomStyle = "margin-bottom: 40px;"
      this.inflectBtnStyle = ""
      this.favoriteBtnStyle = ""
    } else {
      this.bottomStyle = ""
      this.inflectBtnStyle = ""
      this.favoriteBtnStyle = ""
    }
    this.wordText = this.word || ""
    this.phoneticText = this.phonetic ? "/" + this.phonetic + "/" : ""
    this.translationText = this.translation || ""
    this.tagText = this.tag || ""
    this.wordMarqueeThreshold = this.resolveWordMarqueeThreshold(sw)
    this.applyWordMarquee(this.wordText)
    const tagLines = this.tagText ? this.tagText.split(" ") : []
    const wrapW = this.contentWidth - 24
    this.translationLines = this.wrapText(this.translationText, wrapW)
    if (this.translationLines.length === 0) {
      this.translationLines = [""]
    }
    this.displayLines = []
    for (let i = 0; i < this.translationLines.length; i++) {
      this.displayLines.push({ text: this.translationLines[i], isTag: false })
    }
    for (let i = 0; i < tagLines.length; i++) {
      this.displayLines.push({ text: tagLines[i], isTag: true })
    }
    this.refreshFavoriteState()
  },

  onDestroy() {
    this.destroyed = true
  },

  goBack() {
    router.back()
  },

  resolveWordMarqueeThreshold(sw) {
    if (sw === 192) return 8
    if (sw === 212) return 9
    if (sw === 336) return 16
    if (sw === 432) return 20
    if (sw === 466) return 20
    return 9
  },

  applyWordMarquee(text) {
    if (text && text.length > this.wordMarqueeThreshold) {
      this.wordMarqueeOn = true
      const extra = text.length - this.wordMarqueeThreshold
      this.wordMarqueeSpeed = Math.max(40, Math.min(60, Math.round(40 + extra * 1.5)))
    } else {
      this.wordMarqueeOn = false
      this.wordMarqueeSpeed = 40
    }
  },

  goInflect() {
    if (!this.wordText) {
      return
    }

    const now = Date.now()
    if (now - this.lastInflectAt < this.inflectPressCooldown) {
      this.showToast("操作太快")
      return
    }
    this.lastInflectAt = now

    const depth = this.currentInflectDepth()
    if (depth >= this.maxInflectDepth) {
      this.showToast("最多查三层")
      return
    }

    router.replace({
      uri: "/pages/results",
      params: {
        query: this.wordText,
        searchMode: "inflect",
        inflectDepth: "" + (depth + 1)
      }
    })
  },

  goInflectParam(queryWord) {
    if (!queryWord) {
      return
    }
    router.push({
      uri: "/pages/results",
      params: {
        query: queryWord,
        searchMode: this.searchMode || "",
        inflectDepth: this.inflectDepth || "0"
      }
    })
  },

  currentInflectDepth() {
    const depth = parseInt(this.inflectDepth)
    if (isNaN(depth) || depth < 0) {
      return 0
    }
    return depth
  },

  showToast(message) {
    const now = Date.now()
    if (now - this.lastToastAt < this.toastCooldown) {
      return
    }
    this.lastToastAt = now
    prompt.showToast({
      message: message,
      duration: 1500
    })
  },

  toggleFavorite() {
    if (!this.wordText) {
      return
    }
    const item = this.currentItem()
    storage.get({
      key: "dic_favorites",
      success: (data) => {
        const list = this.parseList(data)
        this.writeFavoriteList(list, item)
      },
      fail: () => {
        this.writeFavoriteList([], item)
      }
    })
  },

  writeFavoriteList(list, item) {
    const word = this.normalizeWord(item.word)
    let existed = false
    const output = []

    for (let i = 0; i < list.length; i++) {
      const oldItem = list[i]
      if (oldItem && this.normalizeWord(oldItem.word) === word) {
        existed = true
      } else if (oldItem && output.length < 20) {
        output.push(oldItem)
      }
    }

    if (!existed) {
      output.unshift(item)
      while (output.length > 20) {
        output.pop()
      }
      this.favoriteLabel = "已收藏"
    } else {
      this.favoriteLabel = "收藏"
    }

    storage.set({
      key: "dic_favorites",
      value: JSON.stringify(output)
    })
  },

  refreshFavoriteState() {
    storage.get({
      key: "dic_favorites",
      success: (data) => {
        if (this.destroyed) return
        const list = this.parseList(data)
        const word = this.normalizeWord(this.wordText)
        this.favoriteLabel = "收藏"
        for (let i = 0; i < list.length; i++) {
          if (list[i] && this.normalizeWord(list[i].word) === word) {
            this.favoriteLabel = "已收藏"
            return
          }
        }
      },
      fail: () => {
        if (this.destroyed) return
        this.favoriteLabel = "收藏"
      }
    })
  },

  currentItem() {
    return {
      word: this.wordText,
      phonetic: this.phonetic || "",
      translation: this.translationText,
      tag: this.tagText
    }
  },

  parseList(data) {
    if (!data) {
      return []
    }
    const text = data.value ? data.value : data
    try {
      const list = JSON.parse(text)
      if (list && list.length) {
        return list
      }
    } catch (err) {
      console.log('[腕上词典] storage data parse failed:', err && err.message)
      return []
    }
    return []
  },

  wrapText(value, maxWidth) {
    const text = (value || "").replace(/\s+/g, " ").trim()
    if (!text) {
      return []
    }

    const lines = []
    let line = ""
    let width = 0
    let lastBreak = -1

    for (let i = 0; i < text.length; i++) {
      const ch = text.charAt(i)
      const chWidth = this.getCharWidth(ch)

      if (width + chWidth > maxWidth && line.length > 0) {
        if (lastBreak > 0) {
          lines.push(line.slice(0, lastBreak).trim())
          line = line.slice(lastBreak).trim() + ch
        } else {
          lines.push(line.trim())
          line = ch
        }
        width = this.measureTextWidth(line)
        lastBreak = this.findLastBreak(line)
      } else {
        line += ch
        width += chWidth
      }

      if (ch === " " || ch === ";" || ch === "," || ch === "，" || ch === "；") {
        lastBreak = line.length
      }
    }

    if (line.trim()) {
      lines.push(line.trim())
    }
    return lines
  },

  measureTextWidth(value) {
    let width = 0
    for (let i = 0; i < value.length; i++) {
      width += this.getCharWidth(value.charAt(i))
    }
    return width
  },

  findLastBreak(value) {
    for (let i = value.length - 1; i >= 0; i--) {
      const ch = value.charAt(i)
      if (ch === " " || ch === ";" || ch === "," || ch === "，" || ch === "；") {
        return i + 1
      }
    }
    return -1
  },

  getCharWidth(ch) {
    if (!ch) {
      return 0
    }
    if (ch >= "\u4e00" && ch <= "\u9fff") {
      return 15
    }
    if (ch >= "A" && ch <= "Z") {
      return 9
    }
    if (ch >= "a" && ch <= "z") {
      return 8
    }
    if (ch >= "0" && ch <= "9") {
      return 8
    }
    if (ch === " " || ch === "." || ch === "," || ch === ";" || ch === ":" || ch === "/" || ch === "-") {
      return 5
    }
    return 9
  },

  normalizeWord(value) {
    return (value || "").toLowerCase().trim()
  },

  keyFor(value) {
    const word = this.normalizeWord(value)
    let key = ""
    for (let i = 0; i < word.length && key.length < 2; i++) {
      const ch = word.charAt(i)
      if ((ch >= "a" && ch <= "z") || (ch >= "0" && ch <= "9")) {
        key += ch
      } else {
        key += "_"
      }
    }
    while (key.length < 2) {
      key += "_"
    }
    return key
  },

  readText(uri, done) {
    if (this.fileCache[uri]) {
      done(this.fileCache[uri])
      return
    }
    file.readText({
      uri: uri,
      encoding: "utf-8",
      success: (data) => {
        if (this.destroyed) return
        const text = data && data.text ? data.text : ""
        this.fileCache[uri] = text
        done(text)
      },
      fail: () => {
        if (this.destroyed) return
        this.fileCache[uri] = ""
        done("")
      }
    })
  },

  loadReverseInflectForms(base, currentWord, showSource) {
    const revShard = this.keyFor(base)
    this.readText("/common/dict/inflect_reverse/ireverse_" + revShard + ".txt", (revText) => {
      if (!revText) {
        return
      }
      const allForms = this.scanReverseInflect(revText, base)
      if (allForms.length > 0) {
        if (showSource) {
          this.sourceWord = base
        }
        this.inflectForms = allForms.filter((f) => f !== currentWord).slice(0, 10)
      }
    })
  },

  onTouchStart(evt) {
    const point = this.getTouchPoint(evt)
    if (point) {
      this.touchStartX = point.clientX
      this.touchStartY = point.clientY
    }
  },

  onTouchEnd(evt) {
    const point = this.getTouchPoint(evt)
    if (!point || this.touchStartX < 0) {
      return
    }

    const endX = point.clientX
    const endY = point.clientY
    const sw = this.$app.$def.data.screenWidth || 212
    const startsInLeftQuarter = this.touchStartX <= sw * 0.25
    const endsAtRightQuarter = endX >= sw * 0.75
    const mostlyHorizontal = Math.abs(endY - this.touchStartY) <= 120

    if (startsInLeftQuarter && endsAtRightQuarter && mostlyHorizontal) {
      router.back()
    }

    this.touchStartX = -1
    this.touchStartY = -1
  },

  getTouchPoint(evt) {
    if (evt && evt.changedTouches && evt.changedTouches.length > 0) {
      return evt.changedTouches[0]
    }
    if (evt && evt.touches && evt.touches.length > 0) {
      return evt.touches[0]
    }
    return null
  }
}
</script>

<template>
  <div class="page" ontouchstart="onTouchStart" ontouchend="onTouchEnd">
    <div class="back-pill" @click="goBack">
      <text class="back-mark">返回</text>
    </div>

      <list-item
        class="record-item"
        type="recordCard"
        for="{{ item in recordItems }}"
        tid="id"
        style="width: {{cardWidth}}px; height: {{ item.itemHeight }}px;"
      >
        <div class="record-card" style="width: {{cardWidth}}px; height: {{ item.cardHeight }}px;" @click="openRecord(item)">
          <div class="record-head">
            <text if="{{ !item.wordMarqueeOn }}" class="record-title">{{ item.title }}</text>
            <text if="{{ item.wordMarqueeOn && !marqueeArmed }}" class="record-title">{{ item.title }}</text>
            <marquee if="{{ item.wordMarqueeOn && marqueeArmed }}" class="record-title-marquee"
              scrollamount="{{ item.wordMarqueeSpeed }}" direction="left" loop="-1" text-offset="40">
              {{ item.title }}   {{ item.title }}
            </marquee>
            <text class="record-tag">{{ item.tag }}</text>
          </div>
          <div class="record-translation-box">
            <text class="record-translation-line" for="{{ line in item.translationLines }}">
              {{ line }}
            </text>
          </div>
        </div>
      </list-item>
    </list>

    <text class="empty-text" if="{{ recordItems.length == 0 }}">{{ emptyText }}</text>
  </div>

<script>
import router from "@system.router"
import storage from "@system.storage"

export default {
  protected: {
    type: ""
  },

  private: {
    recordItems: [],
    emptyText: "",
    touchStartX: -1,
    touchStartY: -1,
    destroyed: false,
    cardWidth: 188,
    contentWidth: 164,
    screenWidth: 0,
    wordMarqueeThreshold: 10,
    marqueeArmed: false
  },

  onInit() {
    const sw = this.$app.$def.data.screenWidth || 212
    this.screenWidth = sw
    this.cardWidth = sw - 20
    this.contentWidth = this.cardWidth - 24
    this.wordMarqueeThreshold = this.resolveWordMarqueeThreshold(sw)
    this.loadRecords()
  },

  onDestroy() {
    this.destroyed = true
  },

  onShow() {
    this.loadRecords()
    // 1.5s 让人先看清单词，再齐步启动所有跑马灯
    setTimeout(() => {
      this.marqueeArmed = true
    }, 1500)
  },

  goBack() {
    router.back()
  },

  resolveWordMarqueeThreshold(sw) {
    if (sw === 192) return 9
    if (sw === 212) return 10
    if (sw === 336) return 20
    if (sw === 432) return 25
    if (sw === 466) return 25
    return 10
  },

  calcCardWordMarqueeSpeed(word) {
    // 速度 ≈ 词长 × 4，让长词追上视觉上的同步循环。除非 word 太长会失控，用 60 封顶。
    const len = (word || "").length
    return Math.min(60, Math.max(40, len * 4))
  },

  canFitWithTag(word, tag) {
    if (!tag) return true
    const WORD_FONT = 28
    const TAG_FONT = 18
    const TAG_MARGIN = 8
    const CARD_PADDING = 24
    const WORD_GLYPH_RATIO = 0.72
    const TAG_GLYPH_RATIO = 0.6
    const SAFETY = 4
    const safeLen = (word || "").length
    if (safeLen === 0) return true
    const wordW = safeLen * WORD_FONT * WORD_GLYPH_RATIO
    const tagW = tag.length * TAG_FONT * TAG_GLYPH_RATIO + TAG_MARGIN + SAFETY
    return wordW + tagW <= this.cardWidth - CARD_PADDING
  },

  highestTag(tag) {
    if (!tag) return ""
    const tags = tag.split(" ")
    const priority = ["zk", "gk", "cet4", "cet6", "ky", "ielts", "toefl", "gre"]
    let best = tags[0] || ""
    let bestIdx = -1
    for (let i = 0; i < tags.length; i++) {
      const idx = priority.indexOf(tags[i])
      if (idx > bestIdx) {
        bestIdx = idx
        best = tags[i]
      }
    }
    return best
  },

  loadRecords() {
    const key = this.getStorageKey()
    this.emptyText = this.type === "favorites" ? "暂无收藏" : "暂无历史"
    storage.get({
      key: key,
      success: (data) => {
        if (this.destroyed) return
        this.recordItems = this.buildRecords(this.parseList(data))
      },
      fail: () => {
        if (this.destroyed) return
        this.recordItems = []
      }
    })
  },

  openRecord(item) {
    if (!item) {
      return
    }
    if (this.type === "history") {
      router.push({
        uri: "/pages/results",
        params: {
          query: item.query || item.word || "",
          searchMode: "",
          inflectDepth: "0"
        }
      })
      return
    }
    if (!item.word) {
      return
    }
    router.push({
      uri: "/pages/detail",
      params: {
        word: item.word,
        phonetic: item.rawPhonetic || "",
        translation: item.translation || "",
        tag: item.rawTag || "",
        searchMode: "",
        inflectDepth: "0"
      }
    })
  },

  getStorageKey() {
    if (this.type === "favorites") {
      return "dic_favorites"
    }
    return "dic_history"
  },

  buildRecords(list) {
    const output = []
    for (let i = 0; i < list.length && output.length < 20; i++) {
      const item = list[i]
      if (!item) {
        continue
      }
      if (this.type === "history") {
        const query = (item.query || "").toString()
        if (!query) {
          continue
        }
        const displayTitle = query
        const visibleTag = ""
        const cardHeight = Math.max(80, 64)
        const titleLen = displayTitle.length
        const wordMarqueeOn = titleLen > this.wordMarqueeThreshold
        const wordMarqueeSpeed = wordMarqueeOn ? this.calcCardWordMarqueeSpeed(displayTitle) : 40
        output.push({
          id: this.normalizeWord(query) + "_" + i,
          title: displayTitle,
          query: query,
          recordKind: "history",
          translationLines: [""],
          rawTag: "",
          tag: visibleTag,
          cardHeight: cardHeight,
          itemHeight: cardHeight + 10,
          wordMarqueeOn: wordMarqueeOn,
          wordMarqueeSpeed: wordMarqueeSpeed
        })
        continue
      }
      const word = (item.word || "").toString()
      if (!word) {
        continue
      }
      const phonetic = item.phonetic || ""
      const translation = item.translation || ""
      const bestTag = this.highestTag(item.tag || "")
      const visibleTag = this.canFitWithTag(word, bestTag) ? bestTag : ""
      const lines = this.wrapText(translation, this.contentWidth)
      const safeLines = lines.length > 0 ? lines : [""]
      const lineHeight = 22
      const cardHeight = Math.max(100, 64 + safeLines.length * lineHeight)
      const titleLen = word.length
      const wordMarqueeOn = titleLen > this.wordMarqueeThreshold
      const wordMarqueeSpeed = wordMarqueeOn ? this.calcCardWordMarqueeSpeed(word) : 40
      output.push({
        id: this.normalizeWord(word) + "_" + i,
        title: word,
        word: word,
        phonetic: phonetic,
        rawPhonetic: phonetic,
        translation: translation,
        translationLines: safeLines,
        recordKind: "favorites",
        tag: visibleTag,
        rawTag: bestTag,
        cardHeight: cardHeight,
        itemHeight: cardHeight + 10,
        wordMarqueeOn: wordMarqueeOn,
        wordMarqueeSpeed: wordMarqueeSpeed
      })
    }
    return output
  },

  wrapText(value, maxWidth) {
    const text = (value || "").replace(/\s+/g, " ").trim()
    if (!text || maxWidth <= 0) {
      return []
    }

    const lines = []
    let line = ""
    let width = 0
    let lastBreak = -1

    for (let i = 0; i < text.length; i++) {
      const ch = text.charAt(i)
      const chWidth = this.getCharWidth(ch)

      if (width + chWidth > maxWidth && line.length > 0) {
        if (lastBreak > 0) {
          lines.push(line.slice(0, lastBreak).trim())
          line = line.slice(lastBreak).trim() + ch
        } else {
          lines.push(line.trim())
          line = ch
        }
        width = this.measureTextWidth(line)
        lastBreak = this.findLastBreak(line)
      } else {
        line += ch
        width += chWidth
      }

      if (ch === " " || ch === ";" || ch === "," || ch === "，" || ch === "；") {
        lastBreak = line.length
      }
    }

    if (line.trim()) {
      lines.push(line.trim())
    }
    return lines
  },

  measureTextWidth(value) {
    let width = 0
    for (let i = 0; i < value.length; i++) {
      width += this.getCharWidth(value.charAt(i))
    }
    return width
  },

  findLastBreak(value) {
    for (let i = value.length - 1; i >= 0; i--) {
      const ch = value.charAt(i)
      if (ch === " " || ch === ";" || ch === "," || ch === "，" || ch === "；") {
        return i + 1
      }
    }
    return -1
  },

  getCharWidth(ch) {
    if (!ch) {
      return 0
    }
    if (ch >= "\u4e00" && ch <= "\u9fff") {
      return 13
    }
    if (ch >= "A" && ch <= "Z") {
      return 8
    }
    if (ch >= "a" && ch <= "z") {
      return 7
    }
    if (ch >= "0" && ch <= "9") {
      return 7
    }
    if (
      ch === " " ||
      ch === "." ||
      ch === "," ||
      ch === ";" ||
      ch === ":" ||
      ch === "/" ||
      ch === "-"
    ) {
      return 4
    }
    return 8
  },

  parseList(data) {
    if (!data) {
      return []
    }
    const text = data.value ? data.value : data
    try {
      const list = JSON.parse(text)
      if (list && list.length) {
        return list
      }
    } catch (err) {
      console.log("[腕上词典] storage data parse failed:", err && err.message)
      return []
    }
    return []
  },

  normalizeWord(value) {
    return (value || "").toLowerCase().trim()
  },

  onTouchStart(evt) {
    const point = this.getTouchPoint(evt)
    if (point) {
      this.touchStartX = point.clientX
      this.touchStartY = point.clientY
    }
  },

  onTouchEnd(evt) {
    const point = this.getTouchPoint(evt)
    if (!point || this.touchStartX < 0) {
      return
    }

    const endX = point.clientX
    const endY = point.clientY
    const sw = this.$app.$def.data.screenWidth || 212
    const startsInLeftQuarter = this.touchStartX <= sw * 0.25
    const endsAtRightQuarter = endX >= sw * 0.75
    const mostlyHorizontal = Math.abs(endY - this.touchStartY) <= 120

    if (startsInLeftQuarter && endsAtRightQuarter && mostlyHorizontal) {
      router.back()
    }

    this.touchStartX = -1
    this.touchStartY = -1
  },

  getTouchPoint(evt) {
    if (evt && evt.changedTouches && evt.changedTouches.length > 0) {
      return evt.changedTouches[0]
    }
    if (evt && evt.touches && evt.touches.length > 0) {
      return evt.touches[0]
    }
    return null
  }
}

  }

    if (this.cache[key] === undefined) return undefined
    var idx = this.order.indexOf(key)
    if (idx >= 0) {
      this.order.splice(idx, 1)
      this.order.push(key)
    }
    return this.cache[key]
  }

  },

  calcCardWordMarqueeSpeed(word) {
    // 速度 ≈ 词长 × 4，让长词追上视觉上的同步循环。除非 word 太长会失控，用 60 封顶。
    const len = (word || "").length
    return Math.min(60, Math.max(40, len * 4))
  },

  goBack() {

  },

  openDetail(item) {
    if (!item || !item.word) {
      return
    }

    router.push({
      uri: "/pages/detail",
      params: {
        word: item.word,
        phonetic: item.rawPhonetic || "",
        translation: item.translation || "",
        tag: item.rawTag || item.tag || "",
        searchMode: this.searchMode || "",
        inflectDepth: this.inflectDepth || "0"
      }
    })
  },

  performSearch() {
    const normalized = this.normalizeWord(this.queryText)
    this.queryText = normalized

    if (!normalized) {
      this.resultCards = []
      this.statusText = ""
      return
    }

    const token = this.searchToken + 1
    this.searchToken = token
    this.resultCards = []
    this.statusText = "搜索中"

    if (this.hasChinese(normalized)) {
      this.collectCn2EnResults(normalized, [], {}, (cnResults) => {
        if (token !== this.searchToken) {
          return
        }
        if (cnResults.length > 0) {
          this.resultCards = cnResults.slice(0, 20)
          this.statusText = cnResults.length >= 20 ? "最多显示20条" : ""
        } else {
          this.performChineseSearch(normalized, token)
        }
      })
      return
    }

    this.loadShardIndex(() => {
      if (token !== this.searchToken) {
        return
      }

      const results = []
      const seen = {}

      if (this.searchMode === "inflect") {
        this.collectInflectResult(normalized, results, seen, () => {
          if (token !== this.searchToken) {
            return
          }

          this.finishEnglishSearch(normalized, token, results)
        })
      } else {
        const shards = this.getCandidateShards(normalized)

        this.collectShardResults(shards, normalized, results, seen, 0, () => {
          this.collectInflectResult(normalized, results, seen, () => {
            if (token !== this.searchToken) {
              return
            }

            this.finishEnglishSearch(normalized, token, results)
          })
        })
      }
    })
  },

  finishEnglishSearch(query, token, results) {
    if (results.length === 0 && this.searchMode !== "inflect" && this.isEnglishWord(query)) {
      this.showToast("未找到，混淆搜索")
      this.statusText = "深度搜索"
      this.collectFuzzyResults(query, (cards) => {
        if (token !== this.searchToken) {
          return
        }
        this.resultCards = cards.slice(0, 20)
        if (this.resultCards.length === 0) {
          this.statusText = "未找到"
        } else if (cards.length >= 20) {
          this.statusText = "相近结果20条"
        } else {
          this.statusText = "相近结果"
        }
      })
      return
    }

    this.resultCards = results.slice(0, 20)
    if (this.resultCards.length === 0) {
      this.statusText = "未找到"
    } else if (results.length >= 20) {
      this.statusText = "最多显示20条"
    } else {
      this.statusText = ""
    }
  },

  performChineseSearch(query, token) {
    const normalizedQuery = this.normalizeChineseText(query)
    if (!normalizedQuery) {
      this.resultCards = []
      this.statusText = ""
      return
    }

    this.loadChineseIds(normalizedQuery, (ids) => {
      if (token !== this.searchToken) {
        return
      }

      const results = []
      const seen = {}
      this.collectChineseResults(ids, normalizedQuery, results, seen, 0, () => {
        if (token !== this.searchToken) {
          return
        }

        this.resultCards = results.slice(0, 20)
        if (this.resultCards.length === 0) {
          this.statusText = "未找到"
        } else if (results.length >= 20) {
          this.statusText = "最多显示20条"
        } else {
          this.statusText = ""
        }
      })
    })
  },

  loadChineseIds(query, done) {
    const firstChar = this.getFirstChineseChar(query)
    if (!firstChar) {
      done([])
      return
    }

    const bucket = this.zhBucketFor(firstChar)
    this.readText("/common/dict/zh_index/zh_" + bucket + ".txt", (text) => {
      const lines = text.split("\n")
      for (let i = 0; i < lines.length; i++) {
        const parts = lines[i].split("\t")
        if (parts.length >= 2 && parts[0] === firstChar) {
          done(this.decodeDeltaIds(parts[1]))
          return
        }
      }
      done([])
    })
  },

  collectChineseResults(ids, query, results, seen, index, done) {
    if (index >= ids.length || results.length >= 20) {
      done()
      return
    }

    const entryId = this.parseEntryId(ids[index])
    if (entryId < 0) {
      this.collectChineseResults(ids, query, results, seen, index + 1, done)
      return
    }

    const shard = this.entryShardFor(entryId)
    this.loadEntryShard(shard, (entries) => {
      let nextIndex = index
      while (nextIndex < ids.length && results.length < 20) {
        const nextId = this.parseEntryId(ids[nextIndex])
        if (nextId < 0 || this.entryShardFor(nextId) !== shard) {
          break
        }

        const item = entries[nextId]
        if (item) {
          const lower = this.normalizeWord(item.word)
          if (!seen[lower] && this.normalizeChineseText(item.translation).indexOf(query) >= 0) {
            seen[lower] = true
            results.push(
              this.createResultCard(lower, item.word, item.phonetic, item.translation, item.tag)
            )
          }
        }
        nextIndex++
      }

      this.collectChineseResults(ids, query, results, seen, nextIndex, done)
    })
  },

  loadCnIndex(bucket, done) {
    const cached = this.cnIndexCache.get(bucket)
    if (cached) {
      done(cached)
      return
    }

    this.readText("/common/dict/cn_index/cn_" + bucket + ".txt", (text) => {
      const indexMap = {}
      const lines = text.split("\n")
      for (let i = 0; i < lines.length; i++) {
        const parts = lines[i].split("\t")
        if (parts.length >= 2) {
          indexMap[parts[0]] = this.decodeDeltaIds(parts[1])
        }
      }
      this.cnIndexCache.set(bucket, indexMap)
      done(indexMap)
    })
  },

  collectCn2EnResults(query, results, seen, done) {
    const firstChar = this.getFirstChineseChar(query)
    if (!firstChar) {
      done(results)
      return
    }

    const bucket = this.zhBucketFor(firstChar)
    this.loadCnIndex(bucket, (indexMap) => {
      let ids = indexMap[query]

      // Prefix match fallback: max 20 candidate keys, 10 IDs per key, 50 total
      if (!ids) {
        const candidates = Object.keys(indexMap)
          .filter((key) => key.indexOf(query) === 0)
          .slice(0, 20)

        const allIds = []
        const idSeen = {}
        for (let i = 0; i < candidates.length; i++) {
          const idList = indexMap[candidates[i]] || []
          const limited = idList.slice(0, 10)
          for (let j = 0; j < limited.length; j++) {
            if (!idSeen[limited[j]]) {
              idSeen[limited[j]] = true
              allIds.push(limited[j])
            }
          }
        }
        ids = allIds.slice(0, 50)
      } else {
        ids = ids.slice(0, 50)
      }

      if (!ids || ids.length === 0) {
        done(results)
        return
      }

      this.fetchEntriesByIds(ids, results, seen, 0, done)
    })
  },

  fetchEntriesByIds(ids, results, seen, index, done) {
    if (index >= ids.length || results.length >= 20) {
      done(results)
      return
    }

    const entryId = this.parseEntryId(ids[index])
    if (entryId < 0) {
      this.fetchEntriesByIds(ids, results, seen, index + 1, done)
      return
    }

    const shard = this.entryShardFor(entryId)
    this.loadEntryShard(shard, (entries) => {
      let nextIndex = index
      while (nextIndex < ids.length && results.length < 20) {
        const nextId = this.parseEntryId(ids[nextIndex])
        if (nextId < 0 || this.entryShardFor(nextId) !== shard) {
          break
        }

        const item = entries[nextId]
        if (item) {
          const lower = this.normalizeWord(item.word)
          if (!seen[lower]) {
            seen[lower] = true
            results.push(
              this.createResultCard(lower, item.word, item.phonetic, item.translation, item.tag)
            )
          }
        }
        nextIndex++
      }

      this.fetchEntriesByIds(ids, results, seen, nextIndex, done)
    })
  },

  loadEntryShard(shard, done) {
    const cached = this.entryCache.get(shard)
    if (cached) {
      done(cached)
      return
    }

    this.readText("/common/dict/entries/entry_" + shard + ".txt", (text) => {
      const entries = {}
      const lines = text.split("\n")
      for (let i = 0; i < lines.length; i++) {
        const parts = lines[i].split("\t")
        if (parts.length >= 4) {
          entries[parts[0]] = {
            word: parts[1],
            phonetic: parts[2],
            translation: parts[3],
            tag: parts[4] || ""
          }
        }
      }
      this.entryCache.set(shard, entries)
      done(entries)
    })
  },

  loadShardIndex(done) {
    if (this.shardIndex) {
      done()
      return
    }

    const index = {}
    for (let i = 0; i < WORD_INDEX_LETTERS.length; i++) {
      const letter = WORD_INDEX_LETTERS.charAt(i)
      index[letter] = [letter]
    }
    this.shardIndex = index
    done()
  },

  getCandidateShards(query) {
    const first = this.normalizeWord(query).charAt(0)
    if (first >= "a" && first <= "z") {
      return [first]
    }
    return []
  },

  collectShardResults(shards, query, results, seen, index, done) {
    if (index >= shards.length || results.length >= 20) {
      done()
      return
    }

    const shard = shards[index]
    this.readText("/common/dict/words/word_" + shard + ".txt", (text) => {
      const candidates = []
      this.collectCompactWordCandidates(text, query, candidates, seen, "", 20 - results.length)
      this.hydrateCompactCandidates(candidates, results, 0, () => {
        this.collectShardResults(shards, query, results, seen, index + 1, done)
      })
    })
  },

  collectInflectResult(query, results, seen, done) {
    if (results.length >= 20 || query.length < 2) {
      done()
      return
    }

    const shard = this.keyFor(query)
    this.readText("/common/dict/inflect/inflect_" + shard + ".txt", (text) => {
      const bases = []
      const lines = text.split("\n")
      for (let i = 0; i < lines.length; i++) {
        const parts = lines[i].split("\t")
        if (parts.length >= 2 && parts[0] === query) {
          const words = parts[1].split(",")
          for (let j = 0; j < words.length; j++) {
            if (words[j]) {
              bases.push(words[j])
            }
          }
          break
        }
      }
      this.collectBaseResults(bases, results, seen, 0, () => {
        this.collectReverseInflectResult(query, results, seen, done)
      })
    })
  },

  collectReverseInflectResult(query, results, seen, done) {
    if (results.length >= 20) {
      done()
      return
    }

    const shard = this.keyFor(query)
    this.readText("/common/dict/inflect_reverse/ireverse_" + shard + ".txt", (text) => {
      const forms = []
      const lines = text.split("\n")
      for (let i = 0; i < lines.length; i++) {
        const parts = lines[i].split("\t")
        if (parts.length >= 2 && parts[0] === query) {
          const words = parts[1].split(",")
          for (let j = 0; j < words.length; j++) {
            const w = words[j]
            if (w && w !== query && !seen[w]) {
              forms.push(w)
            }
          }
          break
        }
      }
      this.collectFormResults(query, forms, results, seen, 0, done)
    })
  },

  collectFormResults(baseWord, forms, results, seen, index, done) {
    if (index >= forms.length || results.length >= 20) {
      done()
      return
    }

    const form = forms[index]
    const lower = this.normalizeWord(form)
    if (!lower || seen[lower]) {
      this.collectFormResults(baseWord, forms, results, seen, index + 1, done)
      return
    }
    seen[lower] = true

    // 先查词库，有则用词库释义；无则用词根释义
    const first = lower.charAt(0)
    if (first >= "a" && first <= "z") {
      this.readText("/common/dict/words/word_" + first + ".txt", (text) => {
        const candidates = []
        this.collectCompactWordCandidates(text, form, candidates, seen, "变形", 1)
        if (candidates.length > 0) {
          this.hydrateCompactCandidates(candidates, results, 0, () => {
            this.collectFormResults(baseWord, forms, results, seen, index + 1, done)
          })
        } else {
          this.hydrateFromBaseWord(baseWord, form, results, () => {
            this.collectFormResults(baseWord, forms, results, seen, index + 1, done)
          })
        }
      })
    } else {
      this.collectFormResults(baseWord, forms, results, seen, index + 1, done)
    }
  },

  hydrateFromBaseWord(baseWord, displayWord, results, done) {
    const first = this.normalizeWord(baseWord).charAt(0)
    if (first < "a" || first > "z") {
      done()
      return
    }
    this.readText("/common/dict/words/word_" + first + ".txt", (text) => {
      const lines = text.split("\n")
      for (let i = 0; i < lines.length; i++) {
        const parts = lines[i].split("\t")
        if (parts.length >= 3 && parts[0].toLowerCase() === baseWord.toLowerCase()) {
          const entryId = this.parseEntryId(parts[1])
          if (entryId >= 0) {
            const shard = this.entryShardFor(entryId)
            this.loadEntryShard(shard, (entries) => {
              const item = entries[entryId]
              if (item) {
                results.push(
                  this.createResultCard(
                    this.normalizeWord(displayWord),
                    displayWord,
                    item.phonetic,
                    item.translation,
                    "变形"
                  )
                )
              }
              done()
            })
            return
          }
          break
        }
      }
      done()
    })
  },

  collectBaseResults(bases, results, seen, index, done) {
    if (index >= bases.length || results.length >= 20) {
      done()
      return
    }

    const base = bases[index]
    const first = this.normalizeWord(base).charAt(0)
    if (first < "a" || first > "z") {
      this.collectBaseResults(bases, results, seen, index + 1, done)
      return
    }

    this.readText("/common/dict/words/word_" + first + ".txt", (text) => {
      const candidates = []
      this.collectCompactWordCandidates(text, base, candidates, seen, "变形", 1)
      this.hydrateCompactCandidates(candidates, results, 0, () => {
        this.collectBaseResults(bases, results, seen, index + 1, done)
      })
    })
  },

  collectCompactWordCandidates(text, query, candidates, seen, source, limit) {
    if (!text || limit <= 0) {
      return
    }
    const normalizedQuery = this.normalizeWord(query)
    const lines = text.split("\n")
    for (let i = 0; i < lines.length && candidates.length < limit; i++) {
      const parts = lines[i].split("\t")
      if (parts.length < 3) {
        continue
      }

      const word = parts[0]
      const lower = this.normalizeWord(word)
      if (!lower || seen[lower]) {
        continue
      }
      if (source) {
        if (lower !== normalizedQuery) {
          continue
        }
      } else if (lower.indexOf(normalizedQuery) !== 0) {
        continue
      }

      const entryId = this.parseEntryId(parts[1])
      if (entryId < 0) {
        continue
      }
      seen[lower] = true
      candidates.push({
        entryId: entryId,
        lower: lower,
        word: word,
        tag: parts[2] || "",
        source: source
      })
    }
  },

  hydrateCompactCandidates(candidates, results, index, done) {
    if (index >= candidates.length || results.length >= 20) {
      done()
      return
    }

    const candidate = candidates[index]
    const shard = this.entryShardFor(candidate.entryId)
    this.loadEntryShard(shard, (entries) => {
      let nextIndex = index
      while (nextIndex < candidates.length && results.length < 20) {
        const next = candidates[nextIndex]
        if (this.entryShardFor(next.entryId) !== shard) {
          break
        }
        const item = entries[next.entryId]
        if (item) {
          results.push(
            this.createResultCard(
              next.lower,
              item.word || next.word,
              item.phonetic,
              item.translation,
              next.source || item.tag || next.tag
            )
          )
        }
        nextIndex++
      }
      this.hydrateCompactCandidates(candidates, results, nextIndex, done)
    })
  },

  collectFuzzyResults(query, done) {
    const shards = this.getFuzzyShards(query)
    const items = []
    const seen = {}
    this.fuzzyWordsChecked = 0
    this.collectFuzzyShardResults(shards, query, items, seen, 0, () => {
      items.sort((left, right) => {
        if (left.distance !== right.distance) {
          return left.distance - right.distance
        }
        if (right.score !== left.score) {
          return right.score - left.score
        }
        return left.word.length - right.word.length
      })

      const candidates = []
      for (let i = 0; i < items.length; i++) {
        candidates.push({
          entryId: items[i].entryId,
          lower: items[i].lower,
          word: items[i].word,
          tag: items[i].tag,
          source: "相近"
        })
      }
      const cards = []
      this.hydrateCompactCandidates(candidates, cards, 0, () => done(cards))
    })
  },

  getFuzzyShards(query) {
    const ordered = []
    const seen = {}
    const first = this.normalizeWord(query).charAt(0)
    if (first >= "a" && first <= "z") {
      this.pushUniqueShards(ordered, seen, [first])
    }
    for (let i = 0; i < WORD_INDEX_LETTERS.length; i++) {
      this.pushUniqueShards(ordered, seen, [WORD_INDEX_LETTERS.charAt(i)])
    }
    return ordered
  },

  pushUniqueShards(output, seen, shards) {
    for (let i = 0; i < shards.length; i++) {
      const shard = shards[i]
      if (shard && !seen[shard]) {
        seen[shard] = true
        output.push(shard)
      }
    }
  },

  collectFuzzyShardResults(shards, query, items, seen, index, done) {
    if (
      index >= shards.length ||
      items.length >= FUZZY_POOL_LIMIT ||
      this.fuzzyWordsChecked >= FUZZY_SCAN_LIMIT
    ) {
      done()
      return
    }

    const shard = shards[index]
    this.readText("/common/dict/words/word_" + shard + ".txt", (text) => {
      this.scanFuzzyWordText(text, query, items, seen)
      this.collectFuzzyShardResults(shards, query, items, seen, index + 1, done)
    })
  },

  scanFuzzyWordText(text, query, items, seen) {
    if (!text) {
      return
    }
    const lines = text.split("\n")
    for (
      let i = 0;
      i < lines.length &&
      items.length < FUZZY_POOL_LIMIT &&
      this.fuzzyWordsChecked < FUZZY_SCAN_LIMIT;
      i++
    ) {
      this.fuzzyWordsChecked++
      const parts = lines[i].split("\t")
      if (parts.length < 3) {
        continue
      }
      const lower = this.normalizeWord(parts[0])
      if (!lower || seen[lower] || Math.abs(lower.length - query.length) > 2) {
        continue
      }
      const distance = this.editDistanceBounded(query, lower, 2)
      if (distance < 0 || distance > 2) {
        continue
      }
      const entryId = this.parseEntryId(parts[1])
      if (entryId < 0) {
        continue
      }
      seen[lower] = true
      items.push({
        entryId: entryId,
        lower: lower,
        word: parts[0],
        tag: parts[2] || "",
        distance: distance,
        score: this.scoreFuzzyCandidate(query, lower, parts[2] || "")
      })
    }
  },

  editDistanceBounded(left, right, limit) {
    const leftLength = left.length
    const rightLength = right.length
    if (Math.abs(leftLength - rightLength) > limit) {
      return -1
    }

    let previous = []
    let current = []
    for (let j = 0; j <= rightLength; j++) {
      previous[j] = j
    }

    for (let i = 1; i <= leftLength; i++) {
      current = [i]
      let rowMin = current[0]
      const from = Math.max(1, i - limit)
      const to = Math.min(rightLength, i + limit)

      for (let j = 1; j < from; j++) {
        current[j] = limit + 1
      }
      for (let j = from; j <= to; j++) {
        const cost = left.charAt(i - 1) === right.charAt(j - 1) ? 0 : 1
        const deletion = previous[j] + 1
        const insertion = current[j - 1] + 1
        const substitution = previous[j - 1] + cost
        const value = Math.min(deletion, insertion, substitution)
        current[j] = value
        if (value < rowMin) {
          rowMin = value
        }
      }
      for (let j = to + 1; j <= rightLength; j++) {
        current[j] = limit + 1
      }
      if (rowMin > limit) {
        return -1
      }
      previous = current
    }

    const result = previous[rightLength]
    return result <= limit ? result : -1
  },

  scoreFuzzyCandidate(query, word, tag) {
    let score = 80 - Math.abs(word.length - query.length) * 8
    const tags = tag || ""
    if (word.charAt(0) === query.charAt(0)) {
      score += 20
    }
    if (tags.indexOf("zk") >= 0) {
      score += 40
    }
    if (tags.indexOf("gk") >= 0) {
      score += 36
    }
    if (tags.indexOf("cet4") >= 0) {
      score += 30
    }
    if (tags.indexOf("cet6") >= 0 || tags.indexOf("ky") >= 0) {
      score += 18
    }
    if (tags.indexOf("toefl") >= 0 || tags.indexOf("ielts") >= 0) {
      score += 8
    }
    if (tags.indexOf("gre") >= 0) {
      score -= 8
    }
    return score
  },

  highestTag(tag) {
    if (!tag) return ""
    const tags = tag.split(" ")
    const priority = ["zk", "gk", "cet4", "cet6", "ky", "ielts", "toefl", "gre"]
    let best = tags[0] || ""
    let bestIdx = -1
    for (let i = 0; i < tags.length; i++) {
      const idx = priority.indexOf(tags[i])
      if (idx > bestIdx) {
        bestIdx = idx
        best = tags[i]
      }
    }
    return best
  },

  createResultCard(id, word, phonetic, translation, tag) {
    const maxW = this.contentWidth
    const lines = this.wrapText(translation || "", maxW)
    const safeLines = lines.length > 0 ? lines : [""]
    const lineHeight = 22
    const cardHeight = Math.max(100, 64 + safeLines.length * lineHeight)
    const bestTag = this.highestTag(tag)
    const showTag = this.canFitWithTag(word || "", bestTag)
    const wordLen = (word || "").length
    const wordMarqueeOn = wordLen > this.wordMarqueeThreshold
    const wordMarqueeSpeed = wordMarqueeOn ? this.calcCardWordMarqueeSpeed(word) : 40

    return {
      id: id,
      word: word,
      phonetic: phonetic ? "/" + phonetic + "/" : "",
      rawPhonetic: phonetic || "",
      translation: translation || "",
      translationLines: safeLines,
      tag: showTag ? bestTag : "",
      rawTag: bestTag,
      cardHeight: cardHeight,
      itemHeight: cardHeight + 10,
      wordMarqueeOn: wordMarqueeOn,
      wordMarqueeSpeed: wordMarqueeSpeed
    }
  },

  canFitWithTag(word, tag) {
    if (!tag) return true
    const WORD_FONT = 28
    const TAG_FONT = 18
    const TAG_MARGIN = 8
    const CARD_PADDING = 24
    const WORD_GLYPH_RATIO = 0.72
    const TAG_GLYPH_RATIO = 0.6
    const SAFETY = 4

    const safeLen = (word || "").length
    if (safeLen === 0) return true
    const wordW = safeLen * WORD_FONT * WORD_GLYPH_RATIO
    const tagW = tag.length * TAG_FONT * TAG_GLYPH_RATIO + TAG_MARGIN + SAFETY
    return wordW + tagW <= this.cardWidth - CARD_PADDING
  },

  wrapText(value, maxWidth) {
    const text = (value || "").replace(/\s+/g, " ").trim()
    if (!text) {
      return []
    }

    const lines = []
    let line = ""
    let width = 0
    let lastBreak = -1

    for (let i = 0; i < text.length; i++) {
      const ch = text.charAt(i)
      const chWidth = this.getCharWidth(ch)

      if (width + chWidth > maxWidth && line.length > 0) {
        if (lastBreak > 0) {
          lines.push(line.slice(0, lastBreak).trim())
          line = line.slice(lastBreak).trim() + ch
        } else {
          lines.push(line.trim())
          line = ch
        }
        width = this.measureTextWidth(line)
        lastBreak = this.findLastBreak(line)
      } else {
        line += ch
        width += chWidth
      }

      if (ch === " " || ch === ";" || ch === "," || ch === "，" || ch === "；") {
        lastBreak = line.length
      }
    }

    if (line.trim()) {
      lines.push(line.trim())
    }
    return lines
  },

  measureTextWidth(value) {
    let width = 0
    for (let i = 0; i < value.length; i++) {
      width += this.getCharWidth(value.charAt(i))
    }
    return width
  },

  findLastBreak(value) {
    for (let i = value.length - 1; i >= 0; i--) {
      const ch = value.charAt(i)
      if (ch === " " || ch === ";" || ch === "," || ch === "，" || ch === "；") {
        return i + 1
      }
    }
    return -1
  },

  getCharWidth(ch) {
    if (!ch) {
      return 0
    }
    if (ch >= "\u4e00" && ch <= "\u9fff") {
      return 13
    }
    if (ch >= "A" && ch <= "Z") {
      return 8
    }
    if (ch >= "a" && ch <= "z") {
      return 7
    }
    if (ch >= "0" && ch <= "9") {
      return 7
    }
    if (
      ch === " " ||
      ch === "." ||
      ch === "," ||
      ch === ";" ||
      ch === ":" ||
      ch === "/" ||
      ch === "-"
    ) {
      return 4
    }
    return 8
  },

  readText(uri, done) {
    const cached = this.fileCache.get(uri)
    if (cached !== undefined) {
      done(cached)
      return
    }

    file.readText({
      uri: uri,
      encoding: "utf-8",
      success: (data) => {
        if (this.destroyed) return
        const text = data && data.text ? data.text : ""
        this.fileCache.set(uri, text)
        done(text)
      },
      fail: () => {
        if (this.destroyed) return
        this.fileCache.set(uri, "")
        done("")
      }
    })
  },

  normalizeWord(value) {
    return (value || "").toLowerCase().trim()
  },

  isEnglishWord(value) {
    return /^[a-z][a-z'-]{0,20}$/.test(value || "")
  },

  showToast(message) {
    const now = Date.now()
    if (now - this.lastToastAt < this.toastCooldown) {
      return
    }
    this.lastToastAt = now
    prompt.showToast({
      message: message,
      duration: 1500
    })
  },

  onDestroy() {
    this.destroyed = true
    this.fileCache = null
    this.entryCache = null
    this.cnIndexCache = null
  },

  hasChinese(value) {
    const text = value || ""
    for (let i = 0; i < text.length; i++) {
      if (this.isChineseChar(text.charAt(i))) {
        return true
      }
    }
    return false
  },

  normalizeChineseText(value) {
    const text = value || ""
    let output = ""
    for (let i = 0; i < text.length; i++) {
      const ch = text.charAt(i)
      if (this.isChineseChar(ch)) {
        output += ch
      }
    }
    return output
  },

  getFirstChineseChar(value) {
    const text = value || ""
    for (let i = 0; i < text.length; i++) {
      const ch = text.charAt(i)
      if (this.isChineseChar(ch)) {
        return ch
      }
    }
    return ""
  },

  isChineseChar(ch) {
    return ch >= "\u4e00" && ch <= "\u9fff"
  },

  zhBucketFor(ch) {
    return this.padHex(ch.charCodeAt(0) % 64)
  },

  padHex(value) {
    const hex = value.toString(16)
    if (hex.length < 2) {
      return "0" + hex
    }
    return hex
  },

  decodeDeltaIds(value) {
    if (typeof value !== "string") {
      return []
    }

    const normalized = value.trim()
    if (!normalized) {
      return []
    }
    const tokens = normalized.split(",")
    const ids = []
    let currentId = 0
    for (let i = 0; i < tokens.length; i++) {
      const token = tokens[i]
      if (!token || !/^[0-9a-z]+$/.test(token)) {
        return []
      }

      let delta = 0
      for (let j = 0; j < token.length; j++) {
        const ch = token.charAt(j)
        const code = ch.charCodeAt(0)
        const digit = code >= 48 && code <= 57 ? code - 48 : code - 87
        if (digit < 0 || digit >= 36 || delta > Math.floor((MAX_SAFE_ENTRY_ID - digit) / 36)) {
          return []
        }
        delta = delta * 36 + digit
      }

      if (i > 0 && delta === 0) {
        return []
      }
      if (currentId > MAX_SAFE_ENTRY_ID - delta) {
        return []
      }
      currentId += delta
      ids.push(currentId)
    }
    return ids
  },

  parseEntryId(value) {
    if (typeof value === "number") {
      if (value < 0 || value > MAX_SAFE_ENTRY_ID || Math.floor(value) !== value) {
        return -1
      }
      return value
    }
    if (typeof value !== "string" || !/^[0-9]+$/.test(value)) {
      return -1
    }
    const id = parseInt(value, 10)
    if (id < 0 || id > MAX_SAFE_ENTRY_ID || Math.floor(id) !== id) {
      return -1
    }
    return id
  },

  entryShardFor(entryId) {
    const index = Math.floor(entryId / 500)
    if (index < 10) {
      return "0" + index
    }
    return "" + index
  },

  keyFor(value) {
    const word = this.normalizeWord(value)
    let key = ""
    for (let i = 0; i < word.length && key.length < 2; i++) {
      const ch = word.charAt(i)
      if ((ch >= "a" && ch <= "z") || (ch >= "0" && ch <= "9")) {
        key += ch
      } else {
        key += "_"
      }
    }
    while (key.length < 2) {
      key += "_"
    }
    return key
  },

  onTouchStart(evt) {
    const point = this.getTouchPoint(evt)
    if (point) {
      this.touchStartX = point.clientX
      this.touchStartY = point.clientY
    }
  },

  onTouchEnd(evt) {
    const point = this.getTouchPoint(evt)
    if (!point || this.touchStartX < 0) {
      return
    }

    const endX = point.clientX
    const endY = point.clientY
    const sw = this.$app.$def.data.screenWidth || 212
    const startsInLeftQuarter = this.touchStartX <= sw * 0.25
    const endsAtRightQuarter = endX >= sw * 0.75
    const mostlyHorizontal = Math.abs(endY - this.touchStartY) <= 120

    if (startsInLeftQuarter && endsAtRightQuarter && mostlyHorizontal) {
      router.back()
    }

    this.touchStartX = -1
    this.touchStartY = -1
  },

  getTouchPoint(evt) {
    if (evt && evt.changedTouches && evt.changedTouches.length > 0) {
      return evt.changedTouches[0]
    }
    if (evt && evt.touches && evt.touches.length > 0) {
      return evt.touches[0]
    }
    return null
  }
}
</script>

<template>

<script>
import device from "@system.device"
import router from "@system.router"
import storage from "@system.storage"
import prompt from "@system.prompt"
import file from "@system.file"
import {setSuggestionSeed, setSuggestions} from "../../common/suggestionState"

const ENGLISH_SUFFIXES = [
  "s",
  "ed",
  "ing",
  "er",
  "est",
  "ly",
  "tion",
  "ment",
  "ness",
  "able",
  "ful",
  "less",
  "ive",
  "al",
  "ous",
  "ity"
]

export default {
  protected: {
    queryParam: "",
    autoSearch: "",
    searchMode: "",
    inflectDepth: "",
    screenProfile: "pill-standard"
  },

  private: {
    query: "",
    cursorIndex: 0,
    displayText: "",
    englishSeed: "",
    englishSuggestionsText: "",
    keyboardHidden: false,
    keyboardType: "QWERTY",
    screenType: "pill-shaped",
    maxLength: 5,
    vibrateMode: "short",
    lastToastAt: 0,
    toastCooldown: 1500,
    touchStartX: -1,
    touchStartY: -1,
    destroyed: false,
    autoSearchPending: false,
    englishSuggestionCache: null,
    englishSuggestionPending: null,
    englishSuggestionToken: 0,
    marqueeActive: false,
    marqueeSpeed: 40
  },

  onInit() {
    this.englishSuggestionCache = {}
    this.englishSuggestionPending = {}
    const app = this.$app.$def
    this.applyScreenInfo(app.data)
    device.getInfo({
      success: (data) => {
        app.updateScreenInfo(data)
        this.applyScreenInfo(app.data)
        this.maxLength = this.getMaxLength(
          data.screenShape || app.data.screenShape,
          Number(data.screenWidth || app.data.screenWidth || 212)
        )
        this.refreshDisplay()
      },
      fail: (data, code) => {
        console.log("search device.getInfo failed: " + code)
      }
    })
    if (this.queryParam) {
      this.query = this.queryParam
      this.cursorIndex = this.query.length
    }
    this.keyboardHidden = this.autoSearch === "1" ? true : false
    this.refreshDisplay()
    if (this.autoSearch === "1" && this.query) {
      this.autoSearchPending = true
      // 等待 InputMethod 异步初始化完成（ready 事件）后再跳转
      // timeout 兜底，防止 ready 事件过早/过晚触发
      setTimeout(() => {
        if (this.autoSearchPending) {
          this.autoSearchPending = false
          this.performSearch()
        }
      }, 500)
    }
  },

  applyScreenInfo(data) {
    const screenShape = data.screenShape || "pill-shaped"
    const screenWidth = Number(data.screenWidth || 212)
    this.screenType = screenShape
    this.screenProfile = data.screenProfile || "pill-standard"
    this.maxLength = this.getMaxLength(screenShape, screenWidth)
  },

  getMaxLength(screenShape, screenWidth) {
    if (screenShape === "circle") {
      return 12
    }
    if (screenShape === "rect") {
      return screenWidth >= 400 ? 15 : 11
    }
    return 5
  },

  onDestroy() {
    this.destroyed = true
    this.stopMarquee()
  },

  onReady() {
    if (this.autoSearchPending) {
      this.autoSearchPending = false
      this.performSearch()
    }
  },

  toggleKeyboard() {
    this.keyboardHidden = !this.keyboardHidden
    this.refreshDisplay()
  },

  onVisibilityChange(evt) {
    if (evt && evt.detail) {
      this.keyboardHidden = !evt.detail.visible
    }
    this.refreshDisplay()
  },

  onKeyDown(evt) {
    console.log("key down: " + JSON.stringify(evt))
  },

  onDelete() {
    if (this.cursorIndex > 0) {
      this.query = this.query.slice(0, this.cursorIndex - 1) + this.query.slice(this.cursorIndex)
      this.cursorIndex--
    }
    this.refreshDisplay()
  },

  onComplete(evt) {
    if (evt && evt.detail && evt.detail.content) {
      const content = evt.detail.content
      this.query =
        this.query.slice(0, this.cursorIndex) + content + this.query.slice(this.cursorIndex)
      this.cursorIndex += content.length
    }
    this.refreshDisplay()
  },

  clearText() {
    this.query = ""
    this.cursorIndex = 0
    this.refreshDisplay()
  },

  moveLeft() {
    if (this.cursorIndex > 0) {
      this.cursorIndex--
    }
    this.refreshDisplay()
  },

  moveRight() {
    if (this.cursorIndex < this.query.length) {
      this.cursorIndex++
    }
    this.refreshDisplay()
  },

  onSearchButtonClick() {
    if (!this.query) {
      router.back()
      return
    }
    this.performSearch()
  },

  performSearch() {
    const normalized = this.normalizeWord(this.query)
    this.keyboardHidden = true
    this.refreshDisplay()

    if (!normalized) {
      return
    }
    if (!this.validateQuery(normalized)) {
      return
    }

    this.saveSearchHistory(normalized)

    router.push({
      uri: "/pages/results",
      params: {
        query: normalized,
        searchMode: this.searchMode || "",
        inflectDepth: this.inflectDepth || "0"
      }
    })
  },

  validateQuery(query) {
    if (this.searchMode === "inflect") {
      if (!this.isEnglishWord(query)) {
        this.showToast("只能输入英文")
        return false
      }
      return true
    }

    const hasChinese = this.hasChinese(query)
    const hasEnglish = this.hasEnglish(query)
    if (hasChinese && hasEnglish) {
      this.showToast("输入非法")
      return false
    }
    if (hasChinese) {
      if (!this.isChineseQuery(query)) {
        this.showToast("输入非法")
        return false
      }
      return true
    }
    if (!this.isEnglishWord(query)) {
      this.showToast("输入非法")
      return false
    }
    return true
  },

  showToast(message) {
    const now = Date.now()
    if (now - this.lastToastAt < this.toastCooldown) {
      return
    }
    this.lastToastAt = now
    prompt.showToast({
      message: message,
      duration: 1500
    })
  },

  saveSearchHistory(query) {
    const item = {
      query: query
    }
    storage.get({
      key: "dic_history",
      success: (data) => {
        const list = this.parseList(data)
        this.writeHistoryList(list, item)
      },
      fail: () => {
        this.writeHistoryList([], item)
      }
    })
  },

  writeHistoryList(list, item) {
    const output = [item]
    for (let i = 0; i < list.length && output.length < 20; i++) {
      const oldItem = list[i]
      if (oldItem && oldItem.query && this.normalizeWord(oldItem.query) !== item.query) {
        output.push(oldItem)
      }
    }
    storage.set({
      key: "dic_history",
      value: JSON.stringify(output)
    })
  },

  parseList(data) {
    if (!data) {
      return []
    }
    const text = data.value ? data.value : data
    try {
      const list = JSON.parse(text)
      if (list && list.length) {
        return list
      }
    } catch (err) {
      console.log("[腕上词典] storage data parse failed:", err && err.message)
      return []
    }
    return []
  },

  onTouchStart(evt) {
    const point = this.getTouchPoint(evt)
    if (point) {
      this.touchStartX = point.clientX
      this.touchStartY = point.clientY
    }
  },

  onTouchEnd(evt) {
    const point = this.getTouchPoint(evt)
    if (!point || this.touchStartX < 0) {
      return
    }

    const endX = point.clientX
    const endY = point.clientY
    const appData = (this.$app && this.$app.$def && this.$app.$def.data) || {}
    const sw = appData.screenWidth || 212
    const startsInLeftQuarter = this.touchStartX <= sw * 0.25
    const endsAtRightQuarter = endX >= sw * 0.75
    const mostlyHorizontal = Math.abs(endY - this.touchStartY) <= 120

    if (startsInLeftQuarter && endsAtRightQuarter && mostlyHorizontal) {
      router.back()
    }

    this.touchStartX = -1
    this.touchStartY = -1
  },

  getTouchPoint(evt) {
    if (evt && evt.changedTouches && evt.changedTouches.length > 0) {
      return evt.changedTouches[0]
    }
    if (evt && evt.touches && evt.touches.length > 0) {
      return evt.touches[0]
    }
    return null
  },

  refreshDisplay() {
    if (!this.query) {
      this.stopMarquee()
      this.cursorIndex = 0
      this.displayText = ""
      this.englishSeed = ""
      this.englishSuggestionsText = ""
      this.englishSuggestionToken++
      setSuggestionSeed("")
      setSuggestions("")
      return
    }

    if (this.cursorIndex < 0) {
      this.cursorIndex = 0
    }
    if (this.cursorIndex > this.query.length) {
      this.cursorIndex = this.query.length
    }

    if (this.query.length > this.maxLength && this.keyboardHidden) {
      this.startMarquee()
    } else {
      this.stopMarquee()
      this.displayText =
        this.query.slice(0, this.cursorIndex) +
        (this.keyboardHidden ? "" : "_") +
        this.query.slice(this.cursorIndex)
    }
    this.updateEnglishSeed()
  },

  startMarquee() {
    this.marqueeSpeed = this.calcMarqueeSpeed(this.query.length)
    this.displayText = this.query + "   " + this.query
    setTimeout(() => {
      const el = this.$element("marquee-text")
      if (el && el.start) {
        el.start()
      }
    }, 50)
  },

  stopMarquee() {
    const el = this.$element("marquee-text")
    if (el && el.stop) {
      el.stop()
    }
  },

  calcMarqueeSpeed(textLength) {
    const extra = Math.max(0, textLength - this.maxLength)
    // 字符越长滚动越快，范围 40~60 px/tick
    return Math.max(40, Math.min(60, Math.round(40 + extra * 1.5)))
  },

  updateEnglishSeed() {
    const text = this.query.slice(0, this.cursorIndex).toLowerCase()
    const match = text.match(/[a-z][a-z'-]*$/)
    const nextSeed = match && match[0] ? match[0] : ""
    if (nextSeed === this.englishSeed) {
      return
    }
    this.englishSeed = nextSeed
    this.refreshEnglishSuggestions(nextSeed)
  },

  refreshEnglishSuggestions(seed) {
    const normalized = this.normalizeEnglishSeed(seed)
    const suggestionToken = ++this.englishSuggestionToken
    if (!normalized) {
      this.englishSuggestionsText = ""
      setSuggestionSeed("")
      setSuggestions("")
      return
    }

    this.loadEnglishSuggestionSource(normalized.charAt(0), (source) => {
      if (suggestionToken !== this.englishSuggestionToken) {
        return
      }
      this.applyEnglishSuggestions(normalized, source)
    })
  },

  loadEnglishSuggestionSource(letter, done) {
    if (Object.prototype.hasOwnProperty.call(this.englishSuggestionCache, letter)) {
      done(this.englishSuggestionCache[letter])
      return
    }
    if (this.englishSuggestionPending[letter]) {
      this.englishSuggestionPending[letter] = done
      return
    }

    this.englishSuggestionPending[letter] = done

    file.readText({
      uri: "/common/dict/words/word_" + letter + ".txt",
      encoding: "utf-8",
      success: (data) => {
        if (this.destroyed) {
          return
        }
        const text = data && data.text ? data.text : ""
        this.englishSuggestionCache[letter] = text
        const pending = this.englishSuggestionPending[letter]
        delete this.englishSuggestionPending[letter]
        pending(text)
      },
      fail: () => {
        if (this.destroyed) {
          return
        }
        this.englishSuggestionCache[letter] = ""
        const pending = this.englishSuggestionPending[letter]
        delete this.englishSuggestionPending[letter]
        pending("")
      }
    })
  },

  applyEnglishSuggestions(normalized, source) {
    const rows = source ? source.split("\n") : []
    const items = []
    const seen = {}
    this.scanEnglishSuggestionRows(rows, normalized, items, seen)
    const suggestions = this.buildEnglishSuggestions(normalized, items, seen).join("|")
    this.englishSuggestionsText = suggestions
    setSuggestionSeed(normalized)
    setSuggestions(suggestions)
  },

  scanEnglishSuggestionRows(rows, seed, items, seen) {
    for (let i = 0; i < rows.length; i++) {
      const parts = rows[i].split("\t")
      if (parts.length < 3 || !/^(0|[1-9]\d*)$/.test(parts[1])) {
        continue
      }
      const word = this.normalizeEnglishSeed(parts[0])
      if (!word || word === seed || seen[word] || word.indexOf(seed) !== 0) {
        continue
      }
      seen[word] = true
      items.push({
        word: word,
        score: this.scoreEnglishCandidate(seed, word, parts[2] || "")
      })
    }
  },

  buildEnglishSuggestions(seed, items, seen) {
    items.sort((left, right) => {
      if (right.score !== left.score) {
        return right.score - left.score
      }
      return left.word.length - right.word.length
    })
    const output = []
    for (let i = 0; i < items.length && output.length < 30; i++) {
      output.push(items[i].word)
    }
    if (seed.length >= 2) {
      for (let i = 0; i < ENGLISH_SUFFIXES.length && output.length < 30; i++) {
        const suffix = ENGLISH_SUFFIXES[i]
        const combined = seed + suffix
        if (!seen[combined] && output.indexOf(suffix) < 0) {
          output.push(suffix)
        }
      }
    }
    return output
  },

  scoreEnglishCandidate(seed, word, tag) {
    let score = 100 - (word.length - seed.length) * 4
    const tags = tag || ""
    if (tags.indexOf("zk") >= 0) {
      score += 40
    }
    if (tags.indexOf("gk") >= 0) {
      score += 36
    }
    if (tags.indexOf("cet4") >= 0) {
      score += 30
    }
    if (tags.indexOf("cet6") >= 0 || tags.indexOf("ky") >= 0) {
      score += 18
    }
    if (tags.indexOf("toefl") >= 0 || tags.indexOf("ielts") >= 0) {
      score += 8
    }
    if (tags.indexOf("gre") >= 0) {
      score -= 8
    }
    return score
  },

  normalizeEnglishSeed(value) {
    const text = (value || "").toLowerCase()
    const match = text.match(/[a-z][a-z'-]*$/)
    return match && match[0] ? match[0] : ""
  },

  normalizeWord(value) {
    return (value || "").toLowerCase().trim()
  },

  isEnglishWord(value) {
    return /^[a-z][a-z'-]{0,20}$/.test(value || "")
  },

  hasEnglish(value) {
    return /[a-z]/.test(value || "")
  },

  hasChinese(value) {
    const text = value || ""
    for (let i = 0; i < text.length; i++) {
      const ch = text.charAt(i)
      if (ch >= "\u4e00" && ch <= "\u9fff") {
        return true
      }
    }
    return false
  },

  isChineseQuery(value) {
    const text = value || ""
    for (let i = 0; i < text.length; i++) {
      const ch = text.charAt(i)
      if (ch < "\u4e00" || ch > "\u9fff") {
        return false
      }
    }
    return text.length > 0
  }
}
</script>

<template>
  <div class="page" ontouchstart="onTouchStart" ontouchend="onTouchEnd">
    <div class="back-button" @click="goBack">
      <text class="back-text">返回</text>
    </div>

    <img class="sponsor-code" src="/common/sponsor-code.png" />
  </div>
</template>

<script>
import router from "@system.router"

export default {
  private: {
    touchStartX: -1,
    touchStartY: -1
  },

  goBack() {
    router.back()
  },

  onTouchStart(evt) {
    const point = this.getTouchPoint(evt)
    if (point) {
      this.touchStartX = point.clientX
      this.touchStartY = point.clientY
    }
  },

  onTouchEnd(evt) {
    const point = this.getTouchPoint(evt)
    if (!point || this.touchStartX < 0) {
      return
    }

    const endX = point.clientX
    const endY = point.clientY
    const sw = this.$app.$def.data.screenWidth || 212
    const startsInLeftQuarter = this.touchStartX <= sw * 0.25
    const endsAtRightQuarter = endX >= sw * 0.75
    const mostlyHorizontal = Math.abs(endY - this.touchStartY) <= 120

    if (startsInLeftQuarter && endsAtRightQuarter && mostlyHorizontal) {
      router.back()
    }

    this.touchStartX = -1
    this.touchStartY = -1
  },

  getTouchPoint(evt) {
    if (evt && evt.changedTouches && evt.changedTouches.length > 0) {
      return evt.changedTouches[0]
    }
    if (evt && evt.touches && evt.touches.length > 0) {
      return evt.touches[0]
    }
    return null
  }
}
</script>
